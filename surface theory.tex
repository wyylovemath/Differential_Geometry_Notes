\chapter{Surface Theory}

\section{Local Theory of Surfaces}

\subsection{Regular Surface}

\definition{Parametrized Surface}\\
A \textit{parametrized surface} is a function $X: U\subseteq\mathbb{R}^2\to\mathbb{R}^3$ which is differentiable. Furthermore, the image of $X$ is called the \textit{trace} of $X$.

\noindent\textbf{Example}: $X: [0, 2\pi)\times[0, \pi]\to\mathbb{R}^3$ given by
\[
(\theta, \varphi)\longmapsto\begin{pmatrix} r\cos\theta\cdot\sin\varphi\\ r\sin\theta\cdot\sin\varphi\\ r\cos\varphi \end{pmatrix}
\]
gives a sphere $S^2$. But this raises problems:

\noindent\begin{itemize}
\item $\forall\theta\in[0, 2\pi)$, $X(\theta, 0) = (0, 0, r)$, which means that when $\varphi = 0$, $X$ is not bijective.
\item Since $\dfrac{\partial X}{\partial\theta}\bigg|_{(0,0)} = (0,0,0)$, then $\dfrac{\partial X}{\partial\theta}$ and $\dfrac{\partial X}{\partial\varphi}$ are linearly dependent at $(0,0)$, so the Jacobian matrix has rank 1. This means that at $(0,0)$, we can not compute the ``normal vector''.
\end{itemize}

\noindent To fix these problems, we use semi circles to cover $S^2$, since each semi circle can be parametrized by a (unit) open ball in $\mathbb{R}^2$.

\definition{Regular Surface}\\
A subset $S\subseteq\mathbb{R}^3$ is called a \textit{regular surface} if $\forall p\in S$, $\exists$ a neighborhood $V\subseteq\mathbb{R}^3$ of $p$ ($V$ can be seen as an open ball centered at $p$ and radius of some $\varepsilon > 0$), and a map $X: U\to V\cap S$ for some (open connected) $U\subseteq\mathbb{R}^2$, such that the followings hold:

\begin{enumerate}
\item $X$ is differentiable, i.e.: $\forall (u, v)\in U$, $X(u, v) = (x(u,v), y(u,v), z(u,v))$ has partial derivatives of all orders.
\item $X$ is a homeomorphism from $U$ onto $V\cap S$.
\item $X$ is \textit{regular}, i.e.: $\forall q\in U$,
\[
dX_q = \begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q)
\]
is of rank 2.
\end{enumerate}

\noindent\textbf{Example}:

\noindent\begin{itemize}
\item Let $S^2 = \{(x, y, z)\in\mathbb{R}^3 \mid x^2 + y^2 + z^2 = 1\}$. We use 6 coordinate patches to cover $S^2$: Let $U := \{(u, v)\in\mathbb{R}^2 \mid u^2 + v^2 < 1\}$. Consider a map $X_1: U\to V\cap S^2$ given by
\[
(u, v)\longmapsto\left(u, v, \sqrt{1 - u^2 - v^2}\right)
\]
Then \underline{check}:

\begin{enumerate}
\item $u$, $v$, $\sqrt{1-u^2-v^2}$ have infinite partial derivatives.
\item $X_1$ is a bijection, with $X_1^{-1}: (x, y, z)\mapsto(x, y)$ continuous.
\item $\forall (u, v)\in U$, compute
\[
(dX_1)_q = \begin{pmatrix}1 & 0\\ 0 & 1\\ * & \#\end{pmatrix}(q)
\]
where $\begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}$ is of rank 2, so $(dX_1)_q$ is of rank 2.
\end{enumerate}

\noindent Similarly, we can define
\[
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_2}\begin{pmatrix}u\\ v\\ -\sqrt{1-u^2-v^2}\end{pmatrix},\quad
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_3}\begin{pmatrix}u\\ \sqrt{1-u^2-v^2}\\ v\end{pmatrix},\quad
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_4}\begin{pmatrix}u\\ -\sqrt{1-u^2-v^2}\\ v\end{pmatrix}
\]
\[
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_5}\begin{pmatrix}\sqrt{1-u^2-v^2}\\ u\\ v\end{pmatrix},\quad\text{and}\quad
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_6}\begin{pmatrix}-\sqrt{1-u^2-v^2}\\ u\\ v\end{pmatrix}
\]
Then $\forall p\in S^2$, $p$ must lie in (at least) one $V_i\cap S^2$, so we can take $\{X_i: U\to V_i\cap S^2 \mid i = 1, 2, \cdots, 6\}$ to conclude that $S^2$ is a regular surface.
\end{itemize}

\noindent\begin{itemize}
\item Let $f: U\subseteq\mathbb{R}^2\to\mathbb{R}$ be a smooth function. Consider the \textit{graph} of $f$
\[
S := \mathrm{graph}(f) := \left\{\begin{pmatrix}x\\ y\\ z\end{pmatrix}\in\mathbb{R}^3 \ \Bigg|\ \begin{pmatrix}x\\ y\end{pmatrix}\in U,\ z = f(x, y)\right\}
\]
To prove that $S$ is a regular surface, in fact, we only need one coordinate patch $X: U\to S$ given by $(u, v)\mapsto(u, v, f(u, v))$. \underline{Check}:

\begin{enumerate}
\item $u$, $v$, $f(u, v)$ are all differentiable (smooth).
\item $X$ is bijective, with $X^{-1}: (x, y, z)\mapsto(x, y)$ continuous.
\item $dX_q = \begin{pmatrix}1 & 0\\ 0 & 1\\ * & \#\end{pmatrix}(q)$ is of rank 2, $\forall q\in U$.
\end{enumerate}

\noindent Therefore, $S = \mathrm{graph}(f)$ is a regular surface. Furthermore, we may ask, does the converse hold?

\begin{proposition}{Regular Surfaces are Locally Graphs}\\
Any regular surface $S$ is locally a graph, or equivalently, in combination with the above example, we conclude that a surface $S\subseteq\mathbb{R}^3$ is regular iff $S$ is locally a graph of some smooth function.
\end{proposition}

\begin{proof}
It suffices to prove the first statement: $\forall p\in S$, let $X: U\xrightarrow{\cong} V\cap S$ satisfy the three conditions, with $p = X(q)$. By the regularity of $X$,
\[
dX_q = \begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q)
\]
is of rank 2, so at least one of the three matrices below has nonzero determinant:
\[
\begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\end{pmatrix}(q),\quad
\begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q),\quad\text{and}\quad
\begin{pmatrix}\dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q)
\]
WLOG, assume the first one has nonzero determinant, so by the Inverse Function Thm, the function $f{(u,v)} := (x(u,v), y(u,v))$, which is smooth since $X$ and $(x, y, z)\mapsto(x, y)$ are smooth, has a differentiable inverse $g = f^{-1}: (x, y)\mapsto(u(x,y), v(x,y))$ onto some neighborhood of $q$. That is,
\[
X\circ g: (x, y)\longmapsto\left(x, y, z\big(u(x,y), v(x,y)\big)\right)
\]
where $z: \mathbb{R}^2\to\mathbb{R}$ given by $(x, y)\mapsto z\big(u(x,y), v(x,y)\big)$ is the desired graph function, which is smooth, on a neighborhood of $p = X(q)$. 
\end{proof}

\item Let $f: \mathbb{R}^3\to\mathbb{R}$ be a smooth map and let $a\in\mathbb{R}$ be such that $\forall p\in f^{-1}(a)$ with $df_p\neq 0$, i.e.: $a$ is a \textit{regular value} of $f$. Then $f^{-1}(a)\subseteq\mathbb{R}^3$ is a regular surface.


\begin{proof}
$\forall p\in f^{-1}(a)$, assume WLOG that $\dfrac{\partial f}{\partial z}(p)\neq 0$. Consider $F := f - c_a$ where $c_a$ is the constant map. Then $\forall p\in f^{-1}(a)$, $F(p) = f(p) - a = 0$ and $\dfrac{\partial F}{\partial z}(p) = \dfrac{\partial f}{\partial z}(p)\neq 0$. Hence, we can apply the Implicit Function Thm, $\exists\ g: U\subseteq\mathbb{R}^2\to\mathbb{R}$ smooth, s.t.: $F(u, v, g(u,v)) = 0$. Consider
\[
W := \{(u, v, g(u,v)) \mid (u, v)\in U\}
\]
containing $p$ clearly. Since $F(W) = \{0\}$, then $p\in W\subseteq F^{-1}(0)$. Therefore, $\exists$ a neighborhood $V$ of $p$, s.t.:
\[
V\cap f^{-1}(a) = V\cap F^{-1}(0) = W = \{(x, y, z)\ \big|\ (x, y)\in U,\ z = g(x, y)\}
\]
i.e.: $f^{-1}(a)$ is locally a graph of smooth function. 
\end{proof}

\item Consider $\frac{x^2}{a^2}+ \frac{y^2}{b^2}+ \frac{z^2}{c^2} =1$. Let $F= \frac{x^2}{a^2}+ \frac{y^2}{b^2}+ \frac{z^2}{c^2} -1$, which is clearly differentiable. $\forall w= (x,y,z)\in F^{-1} (0)$, since WLOG, $x \neq 0$, then $dF_w \neq 0$, so $0$ is a regular value of $F$. Hence, by above, $F^{-1} (0) = \{(x,y,z) \mid \frac{x^2}{a^2}+ \frac{y^2}{b^2}+ \frac{z^2}{c^2} =1\} \subseteq \mathbb{R}^3$ is a regular surface.
\end{itemize}

That is, define $X: \mathbb{R}^2 \to S^2 \subseteq \mathbb{R}^3$ by $X(u,v) :=$ the intersection of the straight line joining $(u,v,0)$ and $(0,0,2)$. This gives a coordinate patch of $S^2 \setminus \{ (0,0,2)\}$. This gives a way to cover $S^1$, different from the first example of regular surface. Hence, there are many ways to get coordinate patches for a fixed surface.

\begin{proposition}{Change of Coordinates}\\
Let $p\in S$ be in the image of two parametrizations $X: U\to S$, $Y: V\to S$, i.e.: $p\in W = X(U)\cap Y(V)$. Then
\[
\psi := X^{-1}\circ Y: Y^{-1}(W)\longrightarrow X^{-1}(W)
\]
is a diffeomorphism.
\end{proposition}

\begin{proof}
Since $X$ is regular at $p$, then WLOG, we can conclude that the map
\[
(u, v)\xmapsto{\ X\ }\big(x(u,v), y(u,v), z(u,v)\big)\xmapsto{\ \pi\ }\big(x(u,v), y(u,v)\big)
\]
is invertible on some neighborhood of $X^{-1}(p)$. Hence,
\[
\psi = X^{-1}\circ Y = \underbrace{(\pi\circ X)^{-1}}_{\text{differentiable}}\circ\underbrace{(\pi\circ Y)}_{\text{differentiable}}
\]
is differentiable. Similarly, $\psi^{-1}$ is differentiable. Therefore, by the bijectivity of $X, Y$, $\psi$ is a differentiable. 
\end{proof}

\definition{Smooth Function on Regular Surface}\\
Let $S$ be a regular surface and let $f: S\to\mathbb{R}$ be a function. We say $f$ is \textit{smooth} at $p\in S$ if $\exists$ coordinate patch $X: U\to V\cap S$ with $p\in X(U)$, s.t.: $f\circ X$ is smooth at $q = X^{-1}(p)\in U$.

\noindent\textbf{Remark}:
\begin{itemize}
\item In fact, for any parametrization $Y: U'\to V'\cap S$ with $p\in Y(U')$,
\[
f\circ Y = (f\circ X)\circ\big(X^{-1}\circ Y\big)
\]
is smooth on $Y^{-1}\big(X(U)\cap Y(U')\big)$.
\item One can extend this definition:
\begin{itemize}
\item to the function $f: S\to\mathbb{R}^n$, and
\item to $f: S_1\to S_2$ to be \textit{smooth} if $\exists$ $X_1, X_2$ coordinates patches of $S_1$ respectively $S_2$, such that

\[
\begin{tikzcd}
& X_1(U_1) \ar[rr, "f"] & & X_2(U_2) \\
U_1 \ar[ur, "X_1"] \ar[rrrr, "X_2^{-1}\circ f\circ X_1\ \text{is smooth}"] & & & & U_2 \ar[lu, "X_2"']
\end{tikzcd}
\]

\end{itemize}
\end{itemize}

\noindent\textbf{Example}: If $f: \mathbb{R}^3\to\mathbb{R}$ is smooth, then $f|_S: S\to\mathbb{R}$ is smooth, since $\forall X: U\to S\cap V$, $f\circ X: (u,v)\mapsto f\big(x(u,v), y(u,v), z(u,v)\big)$ is a composition of smooth functions.

\begin{proposition}{Double Cone is Not Regular}\\
Consider $F: \mathbb{R}^3\to\mathbb{R}$ given by $(x, y, z)\mapsto x^2 + y^2 - z^2$. Then $F^{-1}(0)$ is not a regular surface.
\end{proposition}

\begin{proof}
\underline{WARNING}: We can not just use $dF_p = (2x, 2y, 2z) = 0$ at $p = (0, 0, 0)$, since perhaps the surface $F^{-1}(0)$ can be described as a regular surface by other parametrizations.

\noindent Suppose on the contrary that $F^{-1}(0)$ is a regular surface. Then a neighborhood of $(0, 0, 0)$ can be described as one of the three functions:
\[
(x, y)\mapsto\big(x, y, f(x, y)\big), \quad (x, z)\mapsto\big(x, g(x, z), z\big), \quad\text{and}\quad (y, z)\mapsto\big(h(y, z), y, z\big)
\]
WLOG, let the first one describe locally $F^{-1}(0)$ at $(0, 0, 0)$. $\forall$ open ball $V = B(0, \varepsilon)$,
\[
(x_0, y_0, z_0)\in V\cap S \iff (x_0, y_0, -z_0)\in V\cap S.
\]
Hence, $f$ is not a function, contradiction. 
\end{proof}


\subsection{Tangent Plane}

\definition{Total Derivative}\\
Let $p\in S$ be in the image of two parametrizations $X_1: U\to V\cap S$, $X_2: \tilde{U}\to\tilde{V}\cap S$. Then the \textit{total derivative} at $p$ is defined as:
\[
d\psi := \begin{pmatrix}\dfrac{\partial\tilde{u}}{\partial u} & \dfrac{\partial\tilde{u}}{\partial v}\\[3mm] \dfrac{\partial\bar{v}}{\partial u} & \dfrac{\partial\bar{v}}{\partial v}\end{pmatrix} = \frac{\partial(\tilde{u}, \bar{v})}{\partial(u, v)}
\]
which is the Jacobian matrix of the transition map $\psi = X_2^{-1}\circ X_1: X_1^{-1}(W)\to X_2(W)$ where $W := X_1(U)\cap X_2(\tilde{U})\ni p$.

\noindent\textbf{Rmk}: By the change of coordinates Prop., $\psi$ is a diffeomorphism, so $\big(d\psi\big)_p$ is invertible.

\begin{proposition}
\[
\frac{\partial X_1}{\partial u} = \frac{\partial\tilde{u}}{\partial u}\frac{\partial X_2}{\partial\tilde{u}} + \frac{\partial\bar{v}}{\partial u}\frac{\partial X_2}{\partial\bar{v}} \quad\text{and}\quad \frac{\partial X_1}{\partial v} = \frac{\partial\tilde{u}}{\partial v}\frac{\partial X_2}{\partial\tilde{u}} + \frac{\partial\bar{v}}{\partial v}\frac{\partial X_2}{\partial\bar{v}}.
\]
\end{proposition}

\begin{proof}
It's immediately from the fact that $\forall(u, v)\in X_1^{-1}(W)$, $X_1(u, v) = X_2\big(\tilde{u}(u, v), \bar{v}(u, v)\big)$.
\end{proof}

\begin{corollary}
\[
\mathrm{Span}\left\{\frac{\partial X_1}{\partial u}, \frac{\partial X_1}{\partial v}\right\} = \mathrm{Span}\left\{\frac{\partial X_2}{\partial\tilde{u}}, \frac{\partial X_2}{\partial\bar{v}}\right\}.
\]
\end{corollary}

\noindent\textbf{Rmk}: This coro. gives the well-defined-ness of tangent plane (we'll introduce it later), i.e.: the tangent plane is not affected by using different coordinate patches.

\definition{Tangent Plane}\\
Let $X: U\to V\cap S$ be a coordinate patch of a regular surface at a point $X(q) = p\in S$. Then the \textit{tangent plane} at $p\in S$ is defined as:
\[
T_pS := \mathrm{Im}\big(dX_p\big) = \mathrm{Span}\left\{\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right\}.
\]

\noindent\textbf{Example}: $S^2$. A coordinate patch is given by $X(u, v) = \big(u, \sqrt{1 - u^2 - v^2}, v\big)$. Then for $p = \left(\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right) = X\left(\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right)$, one has
\[
dX_p = \begin{pmatrix}1 & 0\\ -\frac{u}{\sqrt{1-u^2-v^2}} & -\frac{v}{\sqrt{1-u^2-v^2}}\\ 0 & 1\end{pmatrix}
\Rightarrow T_p\big(S^2\big) = \mathrm{Span}\left\{\begin{pmatrix}1\\ -\frac{1}{\sqrt{3}}\\ 0\end{pmatrix}, \begin{pmatrix}0\\ -\frac{1}{\sqrt{3}}\\ 1\end{pmatrix}\right\} = \mathrm{Span}\left\{\begin{pmatrix}1\\ -1\\ 0\end{pmatrix}, \begin{pmatrix}0\\ -1\\ 1\end{pmatrix}\right\}.
\]

\noindent\begin{itemize}
\item For a (local) graph, $X(u, v) = (u, v, f(u, v))$, $p\in S$ with $p = X(u_0, v_0)$,
\[
T_pS = \mathrm{Span}\left\{\begin{pmatrix}1\\ 0\\ \frac{\partial f}{\partial u}(u_0, v_0)\end{pmatrix}, \begin{pmatrix}0\\ 1\\ \frac{\partial f}{\partial v}(u_0, v_0)\end{pmatrix}\right\}
\]
\end{itemize}
\noindent or $\langle x - p, n\rangle = 0$, where $n$ is normal to $T_pS$.

\definition{Normal Vector}\\
The \textit{normal vector} of a point $p$ on a coordinate patch $X: U\to V\cap S$ is defined as:
\[
N(p) := \frac{\dfrac{\partial X}{\partial u}\times\dfrac{\partial X}{\partial v}}{\big\|\dfrac{\partial X}{\partial u}\times\dfrac{\partial X}{\partial v}\big\|}\ (q)
\]
where $q := X^{-1}(p)$.

\noindent\textbf{Example}: $S^2$.
\[
N\left(\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right) = \frac{\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}\times\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}}{\big\|\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}\times\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}\big\|} = \left(-\frac{1}{\sqrt{3}}, -\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right)
\]

\noindent\begin{itemize}
\item Local graph:
\[
N(p) = \frac{\left(-\frac{\partial f}{\partial u}, -\frac{\partial f}{\partial v}, 1\right)}{\big\|\left(-\frac{\partial f}{\partial u}, -\frac{\partial f}{\partial v}, 1\right)\big\|}
\]
\end{itemize}

\noindent\textbf{Remark}: Observe that $N(p)$ can be ``$+$'' or ``$-$'' along the same direction, so it may be affected by $X$ up to $\pm 1$. This problem will be fixed by the ``orientability''.

\definition{Tangent Vector}\\
A \textit{tangent vector} at $p\in S$ is the tangent vector $\alpha'(0)$ of a regular curve $\alpha: (-\varepsilon, \varepsilon)\to S\subseteq\mathbb{R}^3$ with $\alpha(0) = p$.

\begin{proposition}
The set of tangent vectors at $p\in S$ is precisely equal to $T_pS$.
\end{proposition}

\begin{proof}
($\subseteq$): Let $\alpha: (-\varepsilon, \varepsilon)\to S$ be a regular curve with $\alpha(0) = p$. Since regular surfaces are locally graphs, then $\exists\ Y: U\to V\cap S$, s.t.:
\[
Y(u, v) = (u, v, f(u, v)) \quad\text{or}\quad (u, g(u, v), v) \quad\text{or}\quad \big(h(u, v), u, v\big).
\]
WLOG, let $Y(u, v) = (u, v, f(u, v))$. By taking $\varepsilon$ to be small enough, we may assume $\alpha\big((-\varepsilon, \varepsilon)\big)\subseteq V\cap S$. Now define
\[
\beta = \left((- \varepsilon, \varepsilon)\xrightarrow{\ \alpha\ }V\cap S\xrightarrow{\ Y^{-1}\ }U\subseteq\mathbb{R}^2\right)
\]
where $Y^{-1}$ is differentiable. Hence, $\alpha = Y\circ\beta = X\circ\big(X^{-1}\circ Y\circ\beta\big) = X\circ(\psi\circ\beta)$ where $\alpha, X, \psi$, and $\beta$ are all differentiable. Therefore,
\[
\alpha'(0) = d\alpha_0 = dX_q\circ d(\psi\circ\beta)_0 = \left(\frac{\partial X}{\partial u}, \frac{\partial X}{\partial v}\right)(q)\cdot\begin{pmatrix}a\\ b\end{pmatrix}\in\mathrm{Span}\left\{\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right\} = T_pS.
\]

\noindent ($\supseteq$): $\forall w\in T_pS$, let $w = \left(\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right)\begin{pmatrix}a\\ b\end{pmatrix}$ by the def of tangent plane, for some $a, b\in\mathbb{R}$. Then define $\gamma: (-\varepsilon, \varepsilon)\to\mathbb{R}^2$ by $t\mapsto q + t\begin{pmatrix}a\\ b\end{pmatrix}$ and then define $\alpha := X\circ\gamma$. Then we have:
\[
\alpha(0) := X\big(\gamma(0)\big) = X(q) = p \quad\text{and}\quad \alpha'(0) = dX_q\circ d\gamma_0 := \left(\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right)\cdot\begin{pmatrix}a\\ b\end{pmatrix}\Bigg|_{t=0} = w.
\]
\end{proof}

\noindent\textbf{Example}: Suppose $f: \mathbb{R}^3\to\mathbb{R}$ is a smooth function and $S = f^{-1}(a)$ where $a$ is a regular value of $f$. Then $S$ is a regular surface. Let $\alpha: (-\varepsilon, \varepsilon)\to S$ be any regular curve on $S$, with $\alpha(0) = p$. Then
\[
f\big(\alpha(t)\big)\in f\big(f^{-1}(a)\big) = \{a\} \Rightarrow f\big(\alpha(t)\big) = a, \quad \forall t\in(-\varepsilon, \varepsilon).
\]
Hence, $df_p\circ d\alpha_0 = 0$, i.e.: $df_p\cdot\alpha'(0) = 0$. Therefore, $df_p$ is normal to $T_pS$.

\subsection{First Fundamental Form (f.f.f.)}

\definition{First Fundamental Form (f.f.f.)}\\
The \textit{f.f.f.} of $S$ is given by the quadratic form $I_p(v) := \langle v, v\rangle$ for $v\in T_pS$.

\noindent\textbf{Remark}: Let $X: U\to V\cap S$ be a coordinate patch with $X(q) = p$. Then
\[
\mathcal{B} = \left\{X_u := \frac{\partial X}{\partial u}(q), X_v := \frac{\partial X}{\partial v}(q)\right\}
\]
is a basis of $T_pS$. If we write $E_p := \langle X_u, X_u\rangle_p$, $F_p := \langle X_u, X_v\rangle_p$ and $G_p := \langle X_v, X_v\rangle_p$ (for brevity, we often drop ``$p$''), then $\forall v, w\in T_pS$, $\exists\ a, b, c, d\in\mathbb{R}$, s.t.:
\[
\langle v, w\rangle_p = \langle aX_u + bX_v, cX_u + dX_v\rangle_p
\]
\[
= ac\,\langle X_u, X_u\rangle_p + (bc + ad)\,\langle X_u, X_v\rangle_p + bd\,\langle X_v, X_v\rangle_p
\]
\[
= ac\,E + (bc + ad)\,F + bd\,G
\]
\[
= \begin{pmatrix}a & b\end{pmatrix}\begin{pmatrix}E & F\\ F & G\end{pmatrix}\begin{pmatrix}c\\ d\end{pmatrix}.
\]
Hence, $\forall v\in T_pS$, $I_p(v) = \begin{pmatrix}a & b\end{pmatrix}\begin{pmatrix}E & F\\ F & G\end{pmatrix}\begin{pmatrix}a\\ b\end{pmatrix}$ where $v = aX_u + bX_v$. Therefore, we can retrieve $I_p$ from the matrix
\[
\begin{pmatrix}E & F\\ F & G\end{pmatrix} := \begin{pmatrix}\langle X_u, X_u\rangle_p & \langle X_u, X_v\rangle_p\\ \langle X_v, X_u\rangle_p & \langle X_v, X_v\rangle_p\end{pmatrix}.
\]
We then will often call this matrix the f.f.f. as well.

\noindent\textbf{Example}:
\begin{itemize}
\item \textit{Cylinder}. $X(u, v) = (\cos u, \sin u, v)$, $0 < u < 2\pi$, $v\in\mathbb{R}$.
\[
\Rightarrow X_u = (-\sin u, \cos u, 0) \quad\text{and}\quad X_v = (0, 0, 1).
\]
\[
\Rightarrow E = \langle X_u, X_u\rangle = \sin^2 u + \cos^2 u = 1, \quad F = \langle X_u, X_v\rangle = 0, \quad\text{and}\quad G = \langle X_v, X_v\rangle = 1.
\]
\[
\Rightarrow I_p = \begin{pmatrix}E & F\\ F & G\end{pmatrix} = \begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}.
\]
\item \textit{Plane}. $X(u, v) = x_0 + uW_1 + vW_2 \Rightarrow I_p = \begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}$
\item \textit{Sphere}. $X(u, v) = \big(u, v, \sqrt{1 - u^2 - v^2}\big)$
\[
\Rightarrow X_u = \left(1, 0, \frac{-u}{\sqrt{1 - u^2 - v^2}}\right) \quad\text{and}\quad X_v = \left(0, 1, \frac{-v}{\sqrt{1 - u^2 - v^2}}\right)
\]
\[
\Rightarrow E = 1 + \frac{u^2}{1 - u^2 - v^2} = \frac{1 - v^2}{1 - u^2 - v^2}, \quad F = \frac{uv}{1 - u^2 - v^2}, \quad\text{and}\quad G = \frac{1 - u^2}{1 - u^2 - v^2}
\]
\[
\Rightarrow I_p = \begin{pmatrix}\dfrac{1 - v^2}{1 - u^2 - v^2} & \dfrac{uv}{1 - u^2 - v^2}\\[3mm] \dfrac{uv}{1 - u^2 - v^2} & \dfrac{1 - u^2}{1 - u^2 - v^2}\end{pmatrix}
\]
\end{itemize}

\noindent On the other hand, consider another coordinate patch $\tilde{X}(\theta, \phi) := (\sin\theta\cos\phi, \sin\theta\sin\phi, \cos\theta)$.
\[
\Rightarrow \tilde{X}_\theta = (\cos\theta\cos\phi, \cos\theta\sin\phi, -\sin\theta) \quad\text{and}\quad \tilde{X}_\phi = (-\sin\theta\sin\phi, \sin\theta\cos\phi, 0).
\]
\[
\Rightarrow \tilde{E} = 1, \quad \tilde{F} = 0, \quad\text{and}\quad \tilde{G} = \sin^2\theta \Rightarrow I_p = \begin{pmatrix}1 & 0\\ 0 & \sin^2\theta\end{pmatrix}
\]

\noindent Therefore, it's natural to ask: if we have two parametrizations $X: U\to V\cap S$ and $\tilde{X}: \tilde{U}\to\tilde{V}\cap S$, then how are their f.f.f.s related?

\begin{proposition}
Let $\psi = X^{-1}\circ\tilde{X}$ be the transition map. Then
\[
\begin{pmatrix}\tilde{E} & \tilde{F}\\ \tilde{F} & \tilde{G}\end{pmatrix} = \big(d\psi\big)^t\begin{pmatrix}E & F\\ F & G\end{pmatrix}\big(d\psi\big)
\]
i.e.: the related f.f.f.s are conjugated by the Jacobian matrix of $\psi$.
\end{proposition}

\begin{proof}
\begin{align*}
\begin{pmatrix}\tilde{E} & \tilde{F}\\ \tilde{F} & \tilde{G}\end{pmatrix} &= \left(\frac{\partial\tilde{X}}{\partial\tilde{u}}, \frac{\partial\tilde{X}}{\partial\bar{v}}\right)^{\!\!t}\left(\frac{\partial\tilde{X}}{\partial\tilde{u}}, \frac{\partial\tilde{X}}{\partial\bar{v}}\right) = \big(d\tilde{X}\big)^t\big(d\tilde{X}\big) = \big(dX\circ d\psi\big)^t\big(dX\circ d\psi\big)\\
&= \big(d\psi\big)^t\Big(\big(dX\big)^t\big(dX\big)\Big)\big(d\psi\big) = \big(d\psi\big)^t\begin{pmatrix}E & F\\ F & G\end{pmatrix}\big(d\psi\big).
\end{align*}
\end{proof}

\begin{corollary}
\[
\sqrt{\tilde{E}\tilde{G} - \tilde{F}^2} = \sqrt{EG - F^2}\cdot\left|\det\left(\frac{\partial(u, v)}{\partial(\tilde{u}, \bar{v})}\right)\right|
\]
\end{corollary}

\begin{proof}
It's immediately from the fact that $\psi: (\tilde{u}, \bar{v})\mapsto\big(u(\tilde{u}, \bar{v}), v(\tilde{u}, \bar{v})\big)$ gives the Jacobian matrix
\[
d\psi = \frac{\partial(u, v)}{\partial(\tilde{u}, \bar{v})} \Rightarrow \det\big(d\psi\big) = \det\big((d\psi)^t\big) = \det\left(\frac{\partial(u, v)}{\partial(\tilde{u}, \bar{v})}\right).
\]
\end{proof}

\subsection{Arc-length and Area}

\subsubsection{Arc-length of Regular Curve on Regular Surface}

\begin{proposition}
Let $\alpha: I\to U\subseteq\mathbb{R}^2$ be a regular curve on $U$, let $S\subseteq\mathbb{R}^3$ be a regular surface, and let $X: U\to V\cap S$ be a coordinate patch. Then the curve $\beta = X\circ\alpha: I\to S$ has length
\begin{align*}
    \int_I\big|\beta'(t)\big|\,dt &= \int_I\sqrt{\Big\langle\frac{d(X\circ\alpha)}{dt}, \frac{d(X\circ\alpha)}{dt}\Big\rangle}\ dt = \int_I\sqrt{\big(dX\cdot d\alpha\big)^t\big(dX\cdot d\alpha\big)}\ dt\\
    &= \int_I\sqrt{\big(x'(t)\ y'(t)\big)\big(dX\big)^t dX\begin{pmatrix}x'(t)\\ y'(t)\end{pmatrix}}\ dt = \int_I\sqrt{\big(x'(t)\ y'(t)\big)\,I_p\begin{pmatrix}x'(t)\\ y'(t)\end{pmatrix}}\ dt\\
    &= \int_I\sqrt{E\big(x'(t)\big)^2 + G\big(y'(t)\big)^2 + 2F\,x'(t)\,y'(t)}\ dt
\end{align*}
\end{proposition}

\subsubsection{Area of Regular Surface}

\definition{Area}\\
Let $X: U\to V\cap S$ be a coordinate patch of $S$ and let $R\subseteq S$ be such that $X(Q) = R$ for some $Q\subseteq U$. Then the \textit{area} of $R$ is given by:
\[
\int_Q d\sigma = \int_Q\big|X_u\times X_v\big|\,du\,dv
\]

\noindent\textbf{Remark}: Note that
\[
\big|X_u\times X_v\big|^2 = |X_u|^2|X_v|^2\sin^2\theta = |X_u|^2|X_v|^2\big(1 - \cos^2\theta\big) = |X_u|^2|X_v|^2 - \langle X_u, X_v\rangle^2
\]
Hence, $\big|X_u\times X_v\big| = \sqrt{EG - F^2}$. If $\tilde{X}: \tilde{U}\to\tilde{V}\cap S$ is another coordinate patch of $S$, and $\psi := X^{-1}\circ\tilde{X}: \tilde{Q}\to Q$ is a transition map, then
\[
\int_{\tilde{Q}}\sqrt{\tilde{E}\tilde{G} - \tilde{F}^2}\ d\tilde{u}\,d\tilde{v} = \int_{\tilde{Q}}\sqrt{EG - F^2}\cdot\left|\det\big(\frac{\partial(u, v)}{\partial(\tilde{u}, \tilde{v})}\big)\right|d\tilde{u}\,d\tilde{v} = \int_Q\sqrt{EG - F^2}\,du\,dv
\]
Therefore, the area of $R$ is independent of the choice of coordinate patches.

\noindent\textbf{Example}:
\begin{itemize}
\item \textit{Cylinder}. $X(u, v) = (\cos u, \sin u, v)$, $0 < u < 2\pi$, $v\in(a, b)$. Then $I_p = \begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix} \Rightarrow EG - F^2 = 1$. Hence,
\[
\text{the area of } X(u, v) = \int_a^b\int_0^{2\pi}1\,du\,dv = 2\pi(b - a).
\]
\item \textit{Sphere}. $X(\theta, \phi) = (\sin\theta\cos\phi, \sin\theta\sin\phi, \cos\theta)$ with $0 < \theta\le\pi$, $0 < \phi < 2\pi$ covers the sphere except for poles. Hence,
\[
\text{the area of } S^2 = \int_0^\pi\int_0^{2\pi}\sqrt{\sin^2\theta}\ d\phi\,d\theta = 4\pi.
\]
\item \textit{Torus}. $X(u, v) = \big((R + r\cos v)\cos u, (R + r\cos v)\sin u, r\sin v\big)$, with $0 < u < 2\pi$, $0 < v < 2\pi$. Then
\[
X_u = \big(-(R + r\cos v)\sin u, (R + r\cos v)\cos u, 0\big), \quad X_v = \big(-r\sin v\cos u, -r\sin v\sin u, r\cos v\big)
\]
Hence,
\[
\begin{cases}
E = \langle X_u, X_u\rangle = (R + r\cos v)^2\sin^2 u + (R + r\cos v)^2\cos^2 u = (R + r\cos v)^2\\[2mm]
F = \langle X_u, X_v\rangle = (R + r\cos v)\sin v\cos u\sin u - (R + r\cos v)\sin v\cos u\sin u + 0 = 0\\[2mm]
G = \langle X_v, X_v\rangle = r^2\sin^2 v\cos^2 u + r^2\sin^2 v\sin^2 u + r^2\cos^2 v = r^2
\end{cases}
\]
Therefore,
\[
I_p = \begin{pmatrix}(R + r\cos v)^2 & 0\\ 0 & r^2\end{pmatrix}
\]
so $\sqrt{EG - F^2} = \sqrt{(R + r\cos v)^2\,r^2} = r(R + r\cos v) \Rightarrow \text{Area of } T^2 = \int_0^{2\pi}\int_0^{2\pi}r(R + r\cos v)\,du\,dv = 4\pi^2 Rr.
$
\end{itemize}

\section{Gauss Map}

\subsection{Orientability}

\noindent\textbf{Goal}: We want to define the curvature of a regular surface. Recall that in the case of curve, in particular, of plane curve,
\[
T' = KN \quad\text{and}\quad N' = -KT
\]
Note that we need a \underline{global} concept, normal vector. We deal with this problem by definition:

\definition{Orientable Surface}\\
A surface $S$ is called \textit{orientable} if there is a smooth map $N: S\to S^2$, s.t.: $N(p)\perp T_pS$, $\forall p\in S$.

\noindent\textbf{Remark}: \begin{itemize}
    \item The codomain of the smooth map is $S^2$ instead of $\mathbb{R}^3$, which is because the normal vector is unit.
    \item This map $N$ is called the \textit{Gauss map} of $S$, which is well-defined up to a sign.
\end{itemize}

\noindent\textbf{Example}: 
\noindent\begin{itemize}
\item If $S$ is covered by exactly one coordinate patch $X: U\to S$, then
\[
N(p) = \frac{X_u\times X_v}{\|X_u\times X_v\|}\ (q) \quad\text{where } q = X^{-1}(p)
\]
e.g.: every graph is orientable.
\item Let $f: \mathbb{R}^3\to\mathbb{R}$ be a smooth function, and let $a\in\mathbb{R}$ be a regular value. Then consider $F := f - a$, so $S = F^{-1}(0)$ is a regular surface. Hence,
\[
N(p) := dF_p = \left(\frac{\partial F}{\partial x}, \frac{\partial F}{\partial y}, \frac{\partial F}{\partial z}\right)(p)
\]
\end{itemize}

\subsection{Differential}

\definition{Differential}\\
Let $f: S_1\to S_2$ be a smooth map. The \textit{differential} of $f$ is defined as follows: $\forall w\in T_pS_1$, let $\alpha: (-\varepsilon, \varepsilon)\to S_1$ be a curve on $S_1$ such that $\alpha(0) = p$ and $\alpha'(0) = w$. Then $f\circ\alpha: (-\varepsilon, \varepsilon)\to S_2$ is such that $\big(f\circ\alpha\big)(0) = f(p)$. Hence, we can define well-definedly:
\[
df_p(w) := \big(f\circ\alpha\big)'(0)\in T_{f(p)}S_2.
\]

\begin{proposition}
The image of $dN_p: T_pS\to T_{N(p)}S^2$ is precisely in $T_pS$.
\end{proposition}

\begin{proof}
Let $X_{(u, v)}$ be a coordinate patch of $S$. Consider $\beta: (-\varepsilon, \varepsilon)\to U^\circ$ given by $t\mapsto(t, 0)$ and $\alpha := X\circ\beta$. Then $\alpha(0) = p$ and $\alpha'(0) = X_u$. Then
\[
dN_p(X_u) := \big(N\circ\alpha\big)'(0) := \big(N\circ X\circ\beta\big)'(0) = \frac{d}{dt}(N\circ X)(t_0) := (N\circ X)_u
\]
Similarly, $dN_p(X_v) = \frac{d}{dt}(N\circ X)(t_0) := (N\circ X)_v$. Since $dN_p: T_pS\to T_{N(p)}S^2$ is a linear transformation and $\{X_u, X_v\}$ forms a basis for $T_pS$, then we can get $dN_p$ from $dN_p(X_u) = (N\circ X)_u$ and $dN_p(X_v) = (N\circ X)_v$. Since every vector on the unit sphere $S^2$ is perpendicular to the tangent space at that point, then $T_{N(p)}S^2\perp N(p)$, so
\[
\big\langle dN_p, N_{f(p)}\big\rangle = 0 \Rightarrow \big\langle (N\circ X)_u, N\big\rangle = \big\langle (N\circ X)_v, N\big\rangle = 0.
\]
Hence, $(N\circ X)_u, (N\circ X)_v\in T_pS$. Therefore, $dN_p: T_pS\to T_pS$.
\end{proof}

\noindent\textbf{Remark}: From now on, to simplify, we write $N(u, v)$ instead of $(N\circ X){(u, v)}$.

\noindent\textbf{Example}: \noindent\begin{itemize}
\item $S = P$ is a plane, i.e.: $X_{(u, v)} = x_0 + u\,\underline{w}_1 + v\,\underline{w}_2 \Rightarrow X_u = \underline{w}_1$, $X_v = \underline{w}_2 \Rightarrow N(u, v) = \frac{X_u\times X_v}{\|X_u\times X_v\|} = \frac{\underline{w}_1\times\underline{w}_2}{\|\underline{w}_1\times\underline{w}_2\|} = \text{constant} \Rightarrow N_u = N_v = 0.$ Hence, $dN_p$ is the zero map, i.e.: $N$ no change.
\item $S = S^2$: $X_{(u, v)} = \big(u, v, \sqrt{1 - u^2 - v^2}\big) \Rightarrow X_u = \big(1, 0, \frac{-u}{\sqrt{1 - u^2 - v^2}}\big)$, $X_v = \big(0, 1, \frac{-v}{\sqrt{1 - u^2 - v^2}}\big)\Rightarrow N(u, v) = \frac{X_u\times X_v}{\|X_u\times X_v\|} = \big(u, v, \sqrt{1 - u^2 - v^2}\big) \Rightarrow N_u = X_u, \ N_v = X_v \Rightarrow dN_p = \mathrm{id}_{T_pS}$.

\item \textit{Saddle}: $X_{(u, v)} = (u, v, v^2 - u^2) \Rightarrow X_u = (1, 0, -2u)$, $X_v = (0, 1, 2v)$.

$\Rightarrow X_u\Big|_{(0, 0, 0)} = (1, 0, 0), \quad X_v\Big|_{(0, 0, 0)} = (0, 1, 0) \quad\text{and}\quad N_{(u, v)} = \left(\frac{2u}{\sqrt{4u^2 + 4v^2 + 1}}, \frac{-2v}{\sqrt{4u^2 + 4v^2 + 1}}, \frac{1}{\sqrt{4u^2 + 4v^2 + 1}}\right) \Rightarrow N_u\Big|_{(0, 0, 0)} = (2, 0, 0) \quad\text{and}\quad N_v\Big|_{(0, 0, 0)} = (0, -2, 0) \Rightarrow dN_{(0, 0, 0)}(X_u) = 2X_u\Big|_{(0, 0, 0)} \quad\text{and}\quad dN_{(0, 0, 0)}(X_v) = -2X_v\Big|_{(0, 0, 0)}$.
\end{itemize}

\subsection{Classification of Points of Surface}

\begin{proposition}
The linear operator $dN_p: T_pS\to T_pS$ is a \textit{self-adjoint} operator w.r.t. the inner product $\langle,\rangle: T_pS\times T_pS\to\mathbb{R}$, i.e.: $\langle dN_p(x), y\rangle = \langle x, dN_p(y)\rangle$.
\end{proposition}

\begin{proof}
$\forall w_1, w_2\in T_pS$, $\exists\ a, b, c, d\in\mathbb{R}$, s.t.: $w_1 = aX_u + bX_v$ and $w_2 = cX_u + dX_v$. Then
\[
\big\langle dN_p(w_1), w_2\big\rangle = \langle aN_u + bN_v, cX_u + dX_v\rangle \quad\text{and}\quad \big\langle w_1, dN_p(w_2)\big\rangle = \langle aX_u + bX_v, cN_u + dN_v\rangle.
\]
Since by the symmetry of $\langle,\rangle$, we have: $\langle N_u, X_u\rangle = \langle X_u, N_u\rangle$ and $\langle N_v, X_v\rangle = \langle X_v, N_v\rangle$, then it suffices to show that $\langle N_u, X_v\rangle = \langle X_u, N_v\rangle$ and $\langle N_v, X_u\rangle = \langle X_v, N_u\rangle$. Note that
\[
\frac{\partial}{\partial u}\langle N, X_v\rangle = \frac{\partial}{\partial u}0 = 0 \Rightarrow \langle N_u, X_v\rangle + \langle N, X_{uv}\rangle = 0 \Rightarrow \langle N_u, X_v\rangle = -\langle N, X_{uv}\rangle
\]
and similarly that
\[
\frac{\partial}{\partial v}\langle N, X_u\rangle = 0 \Rightarrow \langle N_v, X_u\rangle = -\langle N, X_{uv}\rangle
\]
Since $X_{uv} = X_{vu}$, then $\langle N_v, X_u\rangle = \langle N_u, X_v\rangle = \langle X_v, N_u\rangle$ and $\langle N_u, X_v\rangle = \langle N_v, X_u\rangle = \langle X_u, N_v\rangle$.
\end{proof}

\definition{Weingarten Operator}\\
The \textit{self-adjoint} operator $-dN_p: T_pS\to T_pS$ is called the \textit{Weingarten operator} or \textit{shape operator}.

\noindent\textbf{Remark}: Let $A: v\mapsto v$ be a self-adjoint operator. Then $\forall i, j$,
\[
a_{ij} = \langle Ae_j, e_i\rangle = \langle e_j, Ae_i\rangle = a_{ji}
\]
where $e_i, e_j$ are standard basis elements. Hence, in real vector space, every self-adjoint operator is symmetric. Since every symmetric matrix is diagonalizable, then every Weingarten operator $-dN_p$ has an orthonormal basis of real eigenvectors.

\definition{Curvature}
\begin{itemize}
\item \textit{Principal curvature}: the real eigenvalues $\lambda_1\le\lambda_2$ of $-dN_p$ are called the \textit{principal curvatures} of $p\in S$. Furthermore, if $\lambda_1 = \lambda_2 = \lambda$, i.e.: $dN_p = \lambda I$, then all directions in $T_pS$ are called \textit{principal directions} of $p\in S$. Otherwise, if $\lambda_1 < \lambda_2$, then only the unit eigenvectors $e_1, e_2\in T_pS$ corresponding to $\lambda_1$ respectively $\lambda_2$ are called the \textit{principal directions} of $p\in S$.
\item \textit{Mean curvature}: The \textit{mean curvature} of $p\in S$ is defined as $H := \frac{1}{2}\big(\lambda_1 + \lambda_2\big) = \frac{1}{2}\mathrm{tr}\big(-dN_p\big)$
\item \textit{Gaussian curvature / scalar curvature}: $K := \lambda_1\lambda_2 = \det(-dN_p)$.
\end{itemize}

\noindent\textbf{Remark}: Note that principal curvatures and mean curvature may differ by a sign by using different $N: S\to S^2$, but principal direction and Gaussian curvature do \textit{not} change.

\noindent\textbf{Example}:\begin{itemize}
\item $S = $ \textit{sphere} of radius $r \Rightarrow dN_p = \begin{pmatrix}-\frac{1}{r} & 0\\ 0 & -\frac{1}{r}\end{pmatrix} \Rightarrow$ principal directions are all directions on $T_pS$ and $H = -\frac{1}{r}$, $K = \frac{1}{r^2}$.
\item $S = $ \textit{cylinder}: $X_{(u, v)} = (a\cos u, a\sin u, v) \Rightarrow X_u = (-a\sin u, a\cos u, 0)$, $X_v = (0, 0, 1) \Rightarrow N_{(u, v)} = (\cos u, \sin u, 0) \Rightarrow N_u = (-\sin u, \cos u, 0)$, $N_v = 0$. Hence, principal directions are: $X_u$, $X_v$, $\lambda_1 = -\frac{1}{a}$, $\lambda_2 = 0 \Rightarrow H = -\frac{1}{2a}$, $K = 0$.
\item \textit{Saddle}: $X_{(u, v)} = (u, v, v^2 - u^2)$, $p = (0, 0, 0) \Rightarrow N_u = 2X_u$, $N_v = -2X_v$, so $\lambda_1 = -2$, $\lambda_2 = 2 \Rightarrow H = 0$, $K = -4$.
\end{itemize}

\definition{Classification of Points of Surface}\\
A point $p\in S$ is called:
\begin{itemize}
\item \textit{elliptic} if $\det(-dN_p) > 0$, e.g.: sphere
\item \textit{hyperbolic} if $\det(-dN_p) < 0$, e.g.: saddle
\item \textit{parabolic} if $\det(-dN_p) = 0$ but $dN_p\neq 0$, e.g.: cylinder
\item \textit{planar} if $dN_p = 0$, e.g.: plane
\end{itemize}

\subsection{Second Fundamental Form}

\definition{Second Fundamental Form}\\
We define a bilinear form $\langle\langle,\rangle\rangle: T_pS\times T_pS\to\mathbb{R}$ by
\[
\langle\langle v, w\rangle\rangle := \big\langle -dN_p(v), w\big\rangle.
\]
The second fundamental form (s.f.f.) $\mathbb{I}_p: T_pS\to\mathbb{R}$ is defined as:
\[
\mathbb{I}_p(w) := \langle\langle w, w\rangle\rangle = \big\langle -dN_p(w), w\big\rangle.
\]

\noindent\textbf{Remark}:
\begin{itemize}
\item \textit{Check}: $\langle\langle v, w\rangle\rangle = \langle\langle w, v\rangle\rangle$ since $-dN_p$ is self-adjoint and $\langle,\rangle$ is linear.
\item \underline{WARNING}: $\langle\langle v, v\rangle\rangle$ can be negative for $v\neq 0$. Hence, $\langle\langle,\rangle\rangle$ is \underline{not} an inner product.
\item How to understand $\mathbb{I}_p$? Suppose $\alpha: (-\varepsilon, \varepsilon)\to S$ is a p.a.l. with $\alpha(0) = p$, $\alpha'(0) = w$. Then we have $T_\alpha'(0) = \alpha''(0) = K_\alpha(0)\,N_\alpha(0). \text{ }\big(\ \underline{\text{WARNING}}:\ N_\alpha(0)\text{ is the normal on }\alpha,$ while $N\big(\alpha(0)\big)$ denotes the normal on $S$, at $p$. In general, they are not the same.$\big)$ Since $\big\langle T_\alpha'(0), T_\alpha(0)\big\rangle = 0$, then
\[
K_\alpha(0)\,N_\alpha(0) = T_\alpha'(0)\in\mathrm{Span}\big\{N\big(\alpha(0)\big),\ N\big(\alpha(0)\big)\times T_\alpha(0) =: T\big\}
\]
i.e.: $K_\alpha N_\alpha = K_nN + K_gT$ for some scalar $K_n$, $K_g$ and hence we can compute:
\[
K_n = \big\langle K_\alpha N_\alpha, N\big\rangle = \big\langle T_\alpha'(0), N\big\rangle = \big\langle\alpha''(0), N\big\rangle
\]
\[
K_g = \big\langle K_\alpha N_\alpha, T\big\rangle = \big\langle T_\alpha'(0), T\big\rangle = \big\langle\alpha''(0), T\big\rangle
\]
\end{itemize}





\definition{Normal Curvature and Geodesic Curvature}\\
$K_n$ is called the \textit{normal curvature} of $\alpha$ on $S$ and $K_g$ is called the \textit{geodesic curvature} of $\alpha$ on $S$.

\noindent\textbf{Remark}: $K_n$ measures how $\alpha$ bend normal $S$ and $K_g$ measures how $\alpha$ bend within $S$.

\noindent\textbf{Example}: Consider a unit vector $w\in T_pS$. Define the \textit{normal section} of $w$ as the plane spanned by $N(p)$ and $w$. Then its intersection with $S$ gives a regular curve $\beta$ p.a.l. with $\beta(0) = p$, $\beta'(0) = w$. This curve $\beta$ is called the \textit{sectional curve} of $w$.
\[
\beta''(0) = T_\beta(0) = K_\beta(0)\,N_\beta(0) = K_nN + K_gT.
\]
Since $\beta$ is a plane curve, then its normal $N_\beta$ must lie in that plane, say, the normal section of $w$. Hence, $K_g = 0$. Therefore,
\[
K_n = \begin{cases}
K_\beta & \text{, if } N_\beta\text{ is along the direction of } N\\[1mm]
-K_\beta & \text{, if } N_\beta\text{ is against the direction of } N.
\end{cases}
\]

\begin{theorem}
Let $w\in T_pS$ be of unit length, and let $\alpha: (-\varepsilon, \varepsilon)\to S$ p.a.l. with $\alpha(0) = p$, and $\alpha'(0) = w$. Then the normal curvature of $\alpha$ on $S$ is equal to $\mathbb{I}_p(w)$.
\end{theorem}

\begin{proof}
\begin{align*}
    \mathbb{I}_p(w) &:= \big\langle -dN_p(w), w\big\rangle = \big\langle -dN_p\big(\alpha'(0)\big), \alpha'(0)\big\rangle = \Big\langle -\big(N\circ\alpha\big)'(0), \alpha'(0)\Big\rangle\\
    &= \Big\langle\big(N\circ\alpha\big)(0), \alpha''(0)\Big\rangle = \Big\langle N(p), K_\alpha(0)\,N_\alpha(0)\Big\rangle := K_n
\end{align*}
\end{proof}

\noindent\textbf{Remark}: This Thm gives the geometric interpretation of $\mathbb{I}_p(w)$.
\begin{itemize}
\item The normal curvature of any $\alpha: (-\varepsilon, \varepsilon)\to S$ with $\alpha'(0) = w$ are the same $\mathbb{I}_p(w)$.
\item $\mathbb{I}_p(w)$ measures the normal curvature of the \textit{sectional curve} of $w$.
\end{itemize}

\begin{corollary}
Let $w\in T_pS$ be of unit length. Then the curvature of the sectional curve of $w$ must lie within $\lambda_1$ and $\lambda_2$, i.e.:
\[
\lambda_1\le\mathbb{I}_p(w)\le\lambda_2 \quad \forall w\in T_pS \text{ with } ||w|| = 1.
\]
\end{corollary}

\begin{proof}
Since $||w|| = 1$, then write $w = \cos\theta\,e_1 + \sin\theta\,e_2$ where $e_1$ and $e_2$ are the principal directions of $-dN_p$ and are orthonormal basis of $T_pS$. Then
\begin{align*}
    \mathbb{I}_p(w) := \big\langle -dN_p(w), w\big\rangle = \Big\langle -dN_p\big(\cos\theta\,e_1 + \sin\theta\,e_2\big), \cos\theta\,e_1 + \sin\theta\,e_2\Big\rangle
    = \lambda_1\cos^2\theta + \lambda_2\sin^2\theta
\end{align*}
Hence, $\lambda_1\le\mathbb{I}_p(w)\le\lambda_2$ and furthermore, the maximum or minimum are obtained when: $\theta = 0$, $\pi$ respectively $\pi/2$, $3\pi/2$.
\end{proof}

\noindent\textbf{Remark}: We can then distinguish the points by:
\begin{itemize}
\item \textit{Elliptic points}: $\lambda_1\le\mathbb{I}_p\le\lambda_2$, $\lambda_1\lambda_2 > 0$.
\item \textit{Hyperbolic points}: $\lambda_1\le\mathbb{I}_p\le\lambda_2$, $\lambda_1\lambda_2 < 0$.
\item \textit{Parabolic points}: $\lambda_1\le\mathbb{I}_p\le\lambda_2$, $\lambda_1\lambda_2 = 0$, $\mathbb{I}_p\neq 0$.
\item \textit{Planar points}: $\mathbb{I}_p = 0$.
\end{itemize}

\definition
Let $N: U\to S^2$ be the normal map w.r.t. $X: U\to V\cap S$. Then we define:
\[
e := -\big\langle N_u, X_u\big\rangle, \quad f := -\big\langle N_u, X_v\big\rangle, \quad\text{and}\quad g := -\big\langle N_v, X_v\big\rangle.
\]

\begin{proposition}
If $w_1 = aX_u + bX_v$, $w_2 = cX_u + dX_v$, then
\[
\big\langle\langle w_1, w_2\rangle\big\rangle = \begin{pmatrix}a & b\end{pmatrix}\begin{pmatrix}e & f\\ f & g\end{pmatrix}\begin{pmatrix}c\\ d\end{pmatrix}.
\]
\end{proposition}

\begin{proposition}
\[
\mathbb{I}_p(w) = \begin{pmatrix}a & b\end{pmatrix}\begin{pmatrix}e & f\\ f & g\end{pmatrix}\begin{pmatrix}a\\ b\end{pmatrix} \quad\text{where } w = aX_u + bX_v.
\]
\end{proposition}

\noindent\textbf{Remark}: 
\begin{itemize}
\item As before, we call $\begin{pmatrix}e & f\\ f & g\end{pmatrix}$ the \textit{Second Fundamental Form} of $p\in S$.
\item But sometimes $N_u$ and $N_v$ are hard to compute, e.g.: Consider a parametrization $X(u, v) = (v\cos u, v\sin u, u)$, $(u, v)\in\mathbb{R}\times(0, \infty)$, of \textit{helicoid}. Then $X_u = (-v\sin u, v\cos u, 1)$ and $X_v = (\cos u, \sin u, 0)$, so
\[
N = \frac{X_u\times X_v}{\|X_u\times X_v\|} = \frac{1}{\sqrt{1 + v^2}}\begin{pmatrix}-\sin u\\ \cos u\\ -v\end{pmatrix}
\]
Hence, we see that $N_u$, $N_v$ are hard to compute. But we have:
\[
\big\langle N, X_u\big\rangle = 0 \Rightarrow \partial_u\big\langle N, X_u\big\rangle = \big\langle N_u, X_u\big\rangle + \big\langle N, X_{uu}\big\rangle = -e + \big\langle N, X_{uu}\big\rangle = 0,
\]
similarly, $0 = -f + \big\langle N, X_{uv}\big\rangle$ and $0 = -g + \big\langle N, X_{vv}\big\rangle$. Then we can compute $e = \big\langle N, X_{uu}\big\rangle$, $f = \big\langle N, X_{uv}\big\rangle$ and $g = \big\langle N, X_{vv}\big\rangle$. We still use the helicoid as the example to see the powers of these formulas:
\[
X_{uu} = (-v\cos u, -v\sin u, 0), \quad X_{uv} = (-\sin u, \cos u, 0), \quad X_{vv} = (0, 0, 0).
\]
Then $e = 0$, $f = \frac{1}{\sqrt{1 + v^2}}$ and $g = 0$
\[
\Rightarrow \mathbb{I}_p = \begin{pmatrix}0 & \frac{1}{\sqrt{1 + v^2}}\\[2mm] \frac{1}{\sqrt{1 + v^2}} & 0\end{pmatrix} \Rightarrow \mathbb{I}_p(w) = \frac{2ab}{\sqrt{1 + v^2}}
\]
for $w = aX_u + bX_v\in T_pS$.
\end{itemize}

\subsection{Weingarten Equation}

\begin{theorem}
\textit{Weingarten equation}. Let $\mathcal{B} = \{X_u, X_v\}$ be a basis for $T_pS$. Then the Weingarten operator $-dN_p: T_pS\to T_pS$ can be computed by
\[
\big(-dN_p\big)_\mathcal{B} = I_p^{-1}\,\mathbb{I}_p.
\]
This formula is called the \textit{Weingarten equation}.
\end{theorem}

\begin{proof}
Let $\theta$ denote the angle between $X_u$ and $X_v$. Since $X_u$ and $X_v$ are linearly independent, then $|\cos\theta| < 1$. Hence,
\begin{align*}
    EG - F^2 &= \big\|X_u\wedge X_v\big\|^2 - \big\langle X_u, X_v\big\rangle^2 = \|X_u\|^2\|X_v\|^2 - \|X_u\|^2\|X_v\|^2\cos^2\theta\\
    &= \|X_u\|^2\|X_v\|^2\big(1 - \cos^2\theta\big) > 0
\end{align*}
i.e.: $I_p$ is always definite positive, so we have the well-defined-ness of $I_p^{-1}$. Now, for proving the formula, suppose $\big(-dN_p\big)_\mathcal{B} = \begin{pmatrix}a & c\\ b & d\end{pmatrix}$, i.e.:
\[
-N_u = -dN_p\big(X_u\big) = aX_u + bX_v \quad\text{and}\quad -N_v = -dN_p\big(X_v\big) = cX_u + dX_v.
\]
Hence,
\[
\begin{cases}
e := \big\langle -N_u, X_u\big\rangle = aE + bF\\[1mm]
f := \big\langle -N_u, X_v\big\rangle = aF + bG \quad\text{or}\quad f := \big\langle -N_v, X_u\big\rangle = cE + dF\\[1mm]
g := \big\langle -N_v, X_v\big\rangle = cF + dG.
\end{cases}
\]
Therefore,
\[
\mathbb{I}_p = \begin{pmatrix}e & f\\ f & g\end{pmatrix} = \begin{pmatrix}E & F\\ F & G\end{pmatrix}\begin{pmatrix}a & c\\ b & d\end{pmatrix} = I_p\big(-dN_p\big)_\mathcal{B}
\]
\end{proof}

\begin{corollary}
The Gaussian curvature $K := \det\big(-dN_p\big) = \det\big(\mathbb{I}_p\big)\big/\det\big(I_p\big) = \dfrac{eg - f^2}{EG - F^2}$ and the mean curvature $H := \frac{1}{2}\mathrm{tr}\big(-dN_p\big) = \frac{1}{2}\cdot\dfrac{eG - 2fF + gE}{EG - F^2}$.
\end{corollary}

\noindent\textbf{Example}:
\begin{itemize}
\item \textit{Helicoid}: $X(u, v) = (v\cos u, v\sin u, u)$. Then
\[
I_p = \begin{pmatrix}1 + v^2 & 0\\ 0 & 1\end{pmatrix} \quad\text{and}\quad \mathbb{I}_p = \begin{pmatrix}0 & \frac{1}{\sqrt{1 + v^2}}\\[2mm] \frac{1}{\sqrt{1 + v^2}} & 0\end{pmatrix}
\]
Hence, by the Weingarten equation,
\[
\big(-dN_p\big)_\mathcal{B} = I_p^{-1}\mathbb{I}_p = \begin{pmatrix}\frac{1}{1 + v^2} & 0\\[2mm] 0 & 1\end{pmatrix}\begin{pmatrix}0 & \frac{1}{\sqrt{1 + v^2}}\\[2mm] \frac{1}{\sqrt{1 + v^2}} & 0\end{pmatrix} = \begin{pmatrix}0 & \big(1 + v^2\big)^{-\frac{3}{2}}\\[2mm] \big(1 + v^2\big)^{-\frac{1}{2}} & 0\end{pmatrix}
\]
and the mean curvature $H := \frac{1}{2}\mathrm{tr}\big(-dN_p\big) = 0$ and the Gaussian curvature $K = \det\big(-dN_p\big) = -\big(1 + v^2\big)^{-2}$.
\item $X(u, v) = (\cosh v\cos u, \cosh v\sin u, v)$ (the \textit{catenoid}). Then $X_u = (-\cosh v\sin u, \cosh v\cos u, 0)$ and $X_v = (\sinh v\cos u, \sinh v\sin u, 1)$, so
\[
E = \cosh^2 v, \quad F = 0, \quad\text{and}\quad G = \sinh^2 v + 1 = \cosh^2 v \Rightarrow I_p = \begin{pmatrix}\cosh^2 v & 0\\ 0 & \cosh^2 v\end{pmatrix}
\]
Furthermore, $X_{uu} = (-\cosh v\cos u, -\cosh v\sin u, 0)$, $X_{uv} = (-\sinh v\sin u, \sinh v\cos u, 0)$, $X_{vv} = (\cosh v\cos u, \cosh v\sin u, 0)$ and $N = \frac{1}{\cosh v}\begin{pmatrix}\cos u\\ \sin u\\ -\sinh v\end{pmatrix}$, so
\[
e = -1, \quad f = 0, \quad g = 1 \Rightarrow \mathbb{I}_p = \begin{pmatrix}-1 & 0\\ 0 & 1\end{pmatrix}
\]
Hence, by the Weingarten equation,
\[
\big(-dN_p\big)_\mathcal{B} = I_p^{-1}\mathbb{I}_p = \frac{1}{\cosh^2 v}\begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}\begin{pmatrix}-1 & 0\\ 0 & 1\end{pmatrix} = \begin{pmatrix}-\frac{1}{\cosh^2 v} & 0\\[2mm] 0 & \frac{1}{\cosh^2 v}\end{pmatrix}
\]
and the mean curvature $H = 0$ and the Gaussian curvature $K = -\frac{1}{\cosh^4 v}$.
\item Consider a parametrization $X(u, v) = \big((a+r\cos u)\cos v, (a+r\cos u)\sin v, r\sin u\big)$ of torus. Then $X_u = (-r\sin u\cos v, -r\sin u\sin v, r\cos u)$ and $X_v = \big(-(a+r\cos u)\sin v, (a+r\cos u)\cos v, 0\big)$, so
\[
E = r^2, \quad F = 0, \quad\text{and}\quad G = (a+r\cos u)^2 \Rightarrow I_p = \begin{pmatrix}r^2 & 0\\ 0 & (a+r\cos u)^2\end{pmatrix}
\]
Furthermore, $X_{uu} = (-r\cos u\cos v, -r\cos u\sin v, -r\sin u)$, $X_{uv} = (r\sin u\sin v, -r\sin u\cos v, 0)$, $X_{vv} = \big(-(a+r\cos u)\cos v, -(a+r\cos u)\sin v, 0\big)$, and $N = (\cos u\cos v, \cos u\sin v, \sin u)$, so
\[
e = r, \quad f = 0, \quad\text{and}\quad g = (a+r\cos u)\cos u \Rightarrow \mathbb{I}_p = \begin{pmatrix}r & 0\\ 0 & (a+r\cos u)\cos u\end{pmatrix}
\]
Hence, the Gaussian curvature
\[
K = \frac{\cos u}{r\,(a+r\cos u)}. \quad\text{Thus,}\quad K\begin{cases}
> 0 & \text{, when } -\frac{\pi}{2} < u < \frac{\pi}{2}\\
= 0 & \text{, when } u = \frac{\pi}{2} \text{ or } \frac{3\pi}{2}\\
< 0 & \text{, when } \frac{\pi}{2} < u < \frac{3\pi}{2}
\end{cases}
\]
Moreover,
\[
\int_{\left[-\frac{\pi}{2}, \frac{3\pi}{2}\right]^2} K\,dA = \int_{-\frac{\pi}{2}}^{\frac{3\pi}{2}}\int_{-\frac{\pi}{2}}^{\frac{3\pi}{2}}\frac{\cos u}{r\,(a+r\cos u)}\cdot r\,(a+r\cos u)\,du\,dv = \int_{-\frac{\pi}{2}}^{\frac{3\pi}{2}}\int_{-\frac{\pi}{2}}^{\frac{3\pi}{2}}\cos u\,du\,dv = 0.
\]
Then we may ask: Does every surface satisfy this condition, i.e.: $\int K\,dA = 0$? The \underline{answer} is \underline{No}: For any ellipsoid $\frac{x^2}{a^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2} = 1$,
\[
\int K\,dA = 4\pi.
\]
\end{itemize}

\subsection{Umbilical Points}

\noindent\textbf{Motivation}: Consider $\lambda^2 - 2H\lambda + K = 0$, whose roots are the principal curvatures $\lambda_1, \lambda_2$, since $\frac{1}{2}(\lambda_1 + \lambda_2) =: H$ and $\lambda_1\lambda_2 =: K$. Hence, $\lambda = H\pm\sqrt{H^2 - K}$, which is a differentiable function w.r.t. $p\in S$, except possible at $H^2 = K$.

\definition{Umbilical Point}\\
A point $p\in S$ is called \textit{umbilical} if $H^2 = K$ at $p$, or equivalently, if $\lambda_1 = \lambda_2$ at $p$.

\begin{theorem}
Let $S$ be a connected regular surface. If $S$ is \textit{totally umbilical}, i.e.: every point on $S$ is umbilical, then $S$ is contained in a sphere or a plane.
\end{theorem}

\begin{proof}
Let $p\in S$ be any point. Then $-dN_p = \begin{pmatrix}\lambda & 0\\ 0 & \lambda\end{pmatrix}_\beta$ for some $\lambda = \lambda(p)$. Thus, $-N_u = -dN_p\big(X_u\big) = \lambda X_u$ and $-N_v = -dN_p\big(X_v\big) = \lambda X_v$. Hence,
\[
-N_{uv} = \lambda_v X_u + \lambda X_{uv} \quad\text{and}\quad -N_{vu} = \lambda_u X_v + \lambda X_{vu}
\]
By the smoothness of $N$ and $X$, we have $N_{uv} = N_{vu}$ and $X_{uv} = X_{vu}$, so $\lambda_v X_u = \lambda_u X_v$. By the linear independence of $\{X_u, X_v\}$, we have $\lambda_u = \lambda_v = 0$. Since $S$ is connected, then $\lambda$ is constant.

\noindent\underline{Case 1}: $\lambda = 0 \Rightarrow -dN_p = 0 \Rightarrow S$ is contained in a plane.

\noindent\underline{Case 2}: $\lambda\neq 0$. Consider $d\Big(X + \frac{1}{\lambda}N\Big)_p = dX_p + \frac{1}{\lambda}dN_p = \begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}_\beta - \frac{1}{\lambda}\begin{pmatrix}\lambda & 0\\ 0 & \lambda\end{pmatrix}_\beta = 0$. Hence, $X + \frac{1}{\lambda}N = $ constant $c$ $\Rightarrow \big\|X - c\big\| = \frac{1}{|\lambda|}$ $\Rightarrow X$ is contained in a sphere centered at $c$ of radius $\frac{1}{|\lambda|}$.
\end{proof}

\section{Third Fundamental Form}

\definition{Third Fundamental Form $\mathbb{I}I_p$}\\
$\forall w_1, w_2\in T_pS$,
\[
\mathbb{I}I_p\big(w_1, w_2\big) := \Big\langle\big(-dN_p\big)^2 w_1,\ w_2\Big\rangle = \big\langle -dN_p w_1,\ -dN_p w_2\big\rangle.
\]

\begin{theorem}
$\mathbb{I}I_p = 2H\mathbb{I}_p - KI_p$.
\end{theorem}

\begin{proof}
By the motivation of umbilical point, we know that $\lambda^2 - 2H\lambda + K = 0$, where $\lambda$ is a principal curvature, i.e.: $-dN_p w = \lambda w$. Hence,
\begin{align*}
    \mathbb{I}I_p(w) &:= \Big\langle -dN_p^2 w,\ w\Big\rangle = \big\langle\lambda^2 w,\ w\big\rangle = \big\langle\big(2H\lambda - K\big)w,\ w\big\rangle\\
    &= 2H\big\langle\lambda w,\ w\big\rangle - K\langle w, w\rangle = \big(2H\mathbb{I}_p - KI_p\big)(w).
\end{align*}
\end{proof}

\definition{N-th Fundamental Form $\mathbb{N}_p$}\\
$\forall w_1, w_2\in T_pS$,
\[
\mathbb{N}_p\big(w_1, w_2\big) := \Big\langle\big(-dN_p\big)^{N - 1} w_1,\ w_2\Big\rangle
\]

\section{Minimal Surface}

\subsection{Definition}

\noindent\textbf{Idea}: Given a coordinate patch $X: U\to V\cap S$, take a closed disc $\overline{D}\subseteq U$. If we ``perturb'' the surface $X\big(\overline{D}\big)$ a little bit, up or down, then we may ask: Will the area go up or down?

\begin{proposition}
Let $\varphi: \overline{D}\times(-\varepsilon, \varepsilon)\to\mathbb{R}^3$ be given by $(u, v, t)\mapsto X(u, v) + th(u, v)N(u, v)$, where $N$ is the normal to $S$ and $h: \overline{D}\to\mathbb{R}$ with $h\big|_{\partial\overline{D}} = 0$ is the ``perturbing function'', and let $X^t(u, v) := \varphi(u, v, t)$. Then
\[
\frac{dA}{dt}\Bigg|_{t = 0} = -\int_D 2hH\sqrt{EG - F^2}\,du\,dv.
\]
\end{proposition}

\begin{proof}
$X_u^t = X_u + th_uN + thN_u$, $X_v^t = X_v + th_vN + thN_v$. Then
\begin{align*}
    E^t &= E + 2th_u\big\langle X_u, N\big\rangle + 2th\big\langle X_u, N_u\big\rangle + t^2h_u^2\langle N, N\rangle + 2t^2h_uh\big\langle N, N_u\big\rangle + t^2h^2\big\langle N_u, N_u\big\rangle\\
    &= E + 2th_u\cdot 0 + 2th\cdot(-e) + t^2h_u^2\cdot 1 + 2t^2h_uh\cdot 0 + t^2h^2\big\langle N_u, N_u\big\rangle\\
    &= E - 2the + O\big(t^2\big)
\end{align*}
and similarly, $F^t = F - 2thf + O\big(t^2\big)$ and $G^t = G - 2thg + O\big(t^2\big)$. Hence,
\begin{align*}
    E^tG^t - \big(F^t\big)^2 &= EG - F^2 - 2th\big(Eg - 2fF + Ge\big) + O\big(t^2\big)\\
    &= EG - F^2 - 4th\big(EG - F^2\big)H + O\big(t^2\big)\\
    &= \big(EG - F^2\big)\big(1 - 4thH\big) + O\big(t^2\big)\\
    &= \big(EG - F^2\big)\big(1 - 4thH + O\big(t^2\big)\big)
\end{align*}
Therefore, $A(t) = \iint_{\overline{D}}\sqrt{E^tG^t - \big(F^t\big)^2}\,du\,dv = \iint_{\overline{D}}\sqrt{EG - F^2}\sqrt{1 - 4thH + O\big(t^2\big)}\,du\,dv$, so
\[
\frac{dA}{dt} = \iint_{\overline{D}}\sqrt{EG - F^2}\,\frac{-2hH + O(t)}{\sqrt{1 - 4thH + O\big(t^2\big)}}\,du\,dv \Rightarrow \frac{dA}{dt}\Bigg|_{t = 0} = -\int_D 2hH\sqrt{EG - F^2}\,du\,dv.
\]
\end{proof}

\begin{corollary}
$H = 0 \iff A'(0) = 0$ $\forall\overline{D}\subseteq U$ and $\forall$ smooth $h$.
\end{corollary}

\begin{proof}
($\Rightarrow$) is obvious. Conversely, suppose on the contrary that $H\neq 0$ at some $p\in\overline{D}$. Then one can choose $h$ such that $h(p) = H(p)\neq 0$. Hence, $A'(0) < 0$, contradicting the assumption.
\end{proof}

\definition{Minimal Surface}\\
A regular parametrized surface is a \textit{minimal surface} if $H = 0$ at every point.

\begin{theorem}
Let $S$ be a minimal surface with parametrization $X(u, v) = \big(u, v, f(u, v)\big)$. Then $f$ satisfies second order quasi-linear PDE
\[
\big(1 + f_u^2\big)f_{vv} - 2f_uf_vf_{uv} + \big(1 + f_v^2\big)f_{uu} = 0.
\]
When $B^2 - 4AC < 0$, this is an elliptic PDE.
\end{theorem}

\begin{proof}
Let $h: \overline{D}\to\mathbb{R}$ be smooth s.t.: $h\big|_{\partial\overline{D}} = 0$. Consider a family $\big\{F^t := f + th\big\}_{t\in(-\varepsilon, \varepsilon)}$ of graphs and then a family $\big\{X^t = (u, v, F^t)\big\}_{t\in(-\varepsilon, \varepsilon)}$ of parametrizations. All have the same boundary. Then compute
\[
0 = \frac{dA^t}{dt}\Bigg|_{t = 0} = \iint_{\overline{D}}\frac{d}{dt}\Bigg|_{t = 0}\sqrt{1 + \big\|\nabla F^t\big\|^2}\,du\,dv = \iint_{\overline{D}}\frac{\big\langle\nabla f, \nabla h\big\rangle}{\sqrt{1 + \|\nabla f\|^2}}\,du\,dv = \iint_{\overline{D}} -h\,\mathrm{div}\left(\frac{\nabla f}{\sqrt{1 + \|\nabla f\|^2}}\right)du\,dv
\]
Since this $h$ is arbitrary, then $-\mathrm{div}\left(\frac{\nabla f}{\sqrt{1 + \|\nabla f\|^2}}\right) = 0$. Then we're done since
\[
-\mathrm{div}\left(\frac{\nabla f}{\sqrt{1 + \|\nabla f\|^2}}\right) = \frac{\big(1 + f_u^2\big)f_{vv} - 2f_uf_vf_{uv} + \big(1 + f_v^2\big)f_{uu}}{\big(1 + \|\nabla f\|^2\big)^{\frac{3}{2}}}.
\]
\end{proof}

\noindent\textbf{Example}: Suppose a solution is of the form $f = \alpha(u) + \beta(v)$. Then by Thm, $\big(1 + \alpha_u^2\big)\beta_{vv} + \big(1 + \beta_v^2\big)\alpha_{uu} = 0$, so
\[
-\frac{1 + \alpha_u^2}{\alpha_{uu}} = \frac{1 + \beta_v^2}{\beta_{vv}}.
\]
Hence, there is $\lambda\in\mathbb{R}$, s.t.:
\[
-\big(1 + \alpha_u^2\big) = \lambda\,\alpha_{uu} \quad\text{and}\quad \big(1 + \beta_v^2\big) = \lambda\,\beta_{vv}.
\]
Therefore,
\[
f(u, v) = \ln\left(\frac{\cos\big(\frac{v}{\lambda}\big)}{\cos\big(\frac{u}{\lambda}\big)}\right)
\]
and the minimal surface $\big(u, v, f(u, v)\big)$ is called the \textit{Scherk's first surface} with $\big(\frac{u}{\lambda}, \frac{v}{\lambda}\big)\in\big(-\frac{\pi}{2}, \frac{\pi}{2}\big)\times\big(-\frac{\pi}{2}, \frac{\pi}{2}\big)$.

\subsection{Isothermal Parametrization}

\definition{Isothermal Parametrization}\\
A parametrization $X(u, v)$ of a surface is called \textit{isothermal} if $E = G$ and $F = 0$, or equivalently, since $I_p$ is positive definite, then $\exists\, e^{\varphi(u, v)}$, s.t.: $I_p = e^{\varphi}\big(du^2 + dv^2\big)$.

\begin{theorem}
Isothermal parametrization exists on minimal surface. In fact, it exists on any smooth (analytic) surface.
\end{theorem}

\begin{proof}
We only prove the ``minimal part'', since the further needs a lot of advanced PDE. Since $S$ is minimal, then $\exists$ smooth $f: \mathbb{R}^2\to\mathbb{R}$, s.t.: $\big(u, v, f(u, v)\big)$ is a parametrization of $S$ and the second order PDE
\[
\big(1 + f_u^2\big)f_{vv} - 2f_uf_vf_{uv} + \big(1 + f_v^2\big)f_{uu} = 0
\]
is satisfied. If we denote the LHS of the PDE equation by $F$, then it can be checked that
\[
\frac{\partial}{\partial v}\left(\frac{1 + f_u^2}{\sqrt{1 + f_u^2 + f_v^2}}\right) - \frac{\partial}{\partial u}\left(\frac{f_uf_v}{\sqrt{1 + f_u^2 + f_v^2}}\right) = \frac{-f_vF}{\sqrt{1 + f_u^2 + f_v^2}} = 0
\]
\[
\frac{\partial}{\partial u}\left(\frac{1 + f_v^2}{\sqrt{1 + f_u^2 + f_v^2}}\right) - \frac{\partial}{\partial v}\left(\frac{f_uf_v}{\sqrt{1 + f_u^2 + f_v^2}}\right) = \frac{-f_uF}{\sqrt{1 + f_u^2 + f_v^2}} = 0
\]
Hence, if we define $Q := \frac{1 + f_u^2}{a}$, $R := \frac{1 + f_v^2}{a}$ and $P = \frac{f_uf_v}{a}$ where $a = \sqrt{1 + f_u^2 + f_v^2}$, then we get: $\frac{\partial Q}{\partial v} - \frac{\partial P}{\partial u} = 0$ and $\frac{\partial R}{\partial u} - \frac{\partial P}{\partial v} = 0$, so the curl of vector fields $V = (Q, P)$ and $W = (P, R)$ are both zero and hence they are conservative. By Green's Thm,
\[
0 = \iint_D\left(\frac{\partial P}{\partial u} - \frac{\partial Q}{\partial v}\right)du\,dv = \int_\gamma Q\,du + P\,dv \quad\text{and}\quad 0 = \iint_D\left(\frac{\partial R}{\partial u} - \frac{\partial P}{\partial v}\right)du\,dv = \int_\gamma P\,du + R\,dv
\]
$\forall$ closed loop $\gamma\in D$. Hence, $\exists$ potentials $\theta$, $\omega$, s.t.: $\nabla\theta = V$, $\nabla\omega = W$. Then consider a linear transformation $T: D\to\mathbb{R}^2$ given by $(u, v)\mapsto(u + \theta, v + \omega)$, whose Jacobian is
\[
\mathcal{J}(T) = \begin{pmatrix}1 + \theta_u & \theta_v\\ \omega_u & 1 + \omega_v\end{pmatrix}
\]
Then $\det\big(\mathcal{J}(T)\big) = \big(1 + \theta_u\big)\big(1 + \omega_v\big) - \theta_v\omega_u = (1 + Q)(1 + R) - P^2 = \frac{(a + 1)^2}{a} > 0$. Hence, we can apply the inverse fct thm: $\exists\, T^{-1}(x, y) = (u, v)$. Then
\[
\begin{pmatrix}u_x & u_y\\ v_x & v_y\end{pmatrix} = \mathcal{J}\big(T^{-1}\big) = \mathcal{J}(T)^{-1} = \frac{1}{\det\big(\mathcal{J}(T)\big)}\begin{pmatrix}1 + \omega_v & -\theta_v\\ -\omega_u & 1 + \theta_u\end{pmatrix} = \frac{a}{(1 + a)^2}\begin{pmatrix}1 + R & -P\\ -P & 1 + Q\end{pmatrix}
\]
Now consider a reparametrization $X(x, y) = \big(u(x, y), v(x, y), f(u(x, y), v(x, y))\big)$ of $S$. Then compute:
\[
X_x = \left(\frac{a + aR}{(1 + a)^2},\ \frac{-aP}{(1 + a)^2},\ f_u\,\frac{a + aR}{(1 + a)^2} + f_v\,\frac{-aP}{(1 + a)^2}\right)
\]
\[
X_y = \left(\frac{-aP}{(1 + a)^2},\ \frac{a + aQ}{(1 + a)^2},\ f_u\,\frac{-aP}{(1 + a)^2} + f_v\,\frac{a + aQ}{(1 + a)^2}\right)
\]
Hence, $E = \big\langle X_x, X_x\big\rangle = \frac{a}{(1 + a)^2} = G$ and $F = 0$, so $X(x, y)$ is isothermal. 
\end{proof}

\begin{theorem}
$\forall\, X(u, v)$ isothermal parametrization of $S$, we have
\[
\Delta X = 2EHN
\]
where $N$ is the normal to $S$.
\end{theorem}

\begin{proof}
Since $X$ is isothermal, then $\big\langle X_u, X_u\big\rangle = \big\langle X_v, X_v\big\rangle$ and $\big\langle X_u, X_v\big\rangle = 0$. Hence,
\[
\frac{\partial}{\partial u}\big\langle X_u, X_u\big\rangle = \frac{\partial}{\partial u}\big\langle X_v, X_v\big\rangle \Rightarrow \big\langle X_{uu}, X_u\big\rangle = \big\langle X_{vu}, X_v\big\rangle
\]
\[
\frac{\partial}{\partial v}\big\langle X_u, X_u\big\rangle = \frac{\partial}{\partial v}\big\langle X_v, X_v\big\rangle \Rightarrow \big\langle X_{uv}, X_u\big\rangle = \big\langle X_{vv}, X_v\big\rangle
\]
\[
\frac{\partial}{\partial u}\big\langle X_u, X_v\big\rangle = 0 \Rightarrow \big\langle X_{uu}, X_v\big\rangle + \big\langle X_u, X_{vu}\big\rangle = 0
\]
\[
\frac{\partial}{\partial v}\big\langle X_u, X_v\big\rangle = 0 \Rightarrow \big\langle X_{uv}, X_v\big\rangle + \big\langle X_u, X_{vv}\big\rangle = 0
\]
Number these from top to bottom by \textcircled{1}, \textcircled{2}, \textcircled{3} and \textcircled{4}. Then
\[
\textcircled{1} + \textcircled{4} \Rightarrow \Big\langle X_{uu} + X_{vv}\ ,\ X_u\Big\rangle = 0
\]
\[
\textcircled{2} + \textcircled{3} \Rightarrow \Big\langle X_{uu} + X_{vv}\ ,\ X_v\Big\rangle = 0
\]
Hence, $\Delta X = X_{uu} + X_{vv}$ is in $N$ direction, so $\Delta X = \lambda N$ for some scalar $\lambda$. Since $X$ is isothermal, then $H = \frac{1}{2}\cdot\frac{eG - 2fF + gE}{EG - F^2} = \frac{e + g}{2E}$. Therefore,
\[
\lambda = \big\langle\Delta X, N\big\rangle = \big\langle X_{uu} + X_{vv}, N\big\rangle = \big\langle X_{uu}, N\big\rangle + \big\langle X_{vv}, N\big\rangle = e + g = 2EH.
\]
\end{proof}

\begin{corollary}
$S$ is a minimal surface with isothermal parametrization $X(u, v)$ iff $\Delta X = 0$, i.e.: $X$ is harmonic.
\end{corollary}

\subsection{Perspective of Complex Analysis}

\noindent\textbf{Recall}:
\begin{itemize}
\item $z = u + iv \Rightarrow u = \frac{z + \bar{z}}{2}$, $v = \frac{z - \bar{z}}{2}$. Then we have:
\[
\begin{cases}
\frac{\partial}{\partial z} = \frac{1}{2}\big(\frac{\partial}{\partial u} - i\,\frac{\partial}{\partial v}\big)\\[2mm]
\frac{\partial}{\partial \bar{z}} = \frac{1}{2}\big(\frac{\partial}{\partial u} + i\,\frac{\partial}{\partial v}\big)
\end{cases}
\]
\item If the complex function $\varphi$ is holomorphic, i.e.: $\varphi$ satisfies the Cauchy-Riemann equation $\frac{\partial u}{\partial x} = \frac{\partial v}{\partial y}$ and $\frac{\partial u}{\partial y} = -\frac{\partial v}{\partial x}$, then $\frac{\partial\varphi}{\partial\bar{z}} = 0$.
\item We write $X(u, v) = X(z, \bar{z})$.
\end{itemize}

\definition
$X(z, \bar{z}) = (x^1, x^2, x^3)$. Let $\varphi = (\varphi^1, \varphi^2, \varphi^3)$ where $\varphi^i := \frac{\partial x^j}{\partial z}$. Then we define:
\[
(\varphi)^2 := \sum_{j=1}^{3}\big(\varphi^j\big)^2 = \sum_{j=1}^{3}\left(\frac{\partial x^j}{\partial z}\right)^2 \quad\text{and}\quad |\varphi|^2 := \sum_{j=1}^{3}\big|\varphi^j\big|^2.
\]

\begin{proposition}
Let $X\big(x^1(u, v), x^2(u, v), x^3(u, v)\big)$ be a parametrization of a surface $S$.
\begin{enumerate}
\item $\varphi^j$ is holomorphic iff $x^j$ is harmonic.
\item $X$ is isothermal iff $(\varphi)^2 = 0$.
\item If $X$ is isothermal, then $S$ is regular iff $|\varphi|^2\neq 0$.
\end{enumerate}
\end{proposition}

\begin{proof}
1.
\begin{align*}
    \frac{\partial\varphi^j}{\partial\bar{z}} &:= \frac{\partial}{\partial\bar{z}}\left(\frac{\partial x^j}{\partial z}\right) = \frac{\partial}{\partial\bar{z}}\left(\frac{1}{2}\big(x_u^j - i\,x_v^j\big)\right) = \frac{1}{2}\left(\frac{\partial x_u^j}{\partial\bar{z}} - i\,\frac{\partial x_v^j}{\partial\bar{z}}\right)\\
    &= \frac{1}{4}\big(x_{uu}^j + i\,x_{uv}^j\big) - \frac{i}{4}\big(x_{vu}^j + i\,x_{vv}^j\big) = \frac{1}{4}\big(x_{uu}^j + x_{vv}^j\big) = \frac{1}{4}\Delta x^j
\end{align*}

\noindent 2.
\begin{align*}
    (\varphi)^2 &:= \left(\frac{\partial x^1}{\partial z}\right)^2 + \left(\frac{\partial x^2}{\partial z}\right)^2 + \left(\frac{\partial x^3}{\partial z}\right)^2\\
    &= \frac{1}{4}\big(x_u^1 - i\,x_v^1\big)^2 + \frac{1}{4}\big(x_u^2 - i\,x_v^2\big)^2 + \frac{1}{4}\big(x_u^3 - i\,x_v^3\big)^2\\
    &= \frac{1}{4}\sum_{j=1}^{3}\left(\big(x_u^j\big)^2 - 2i\,x_u^j\,x_v^j - \big(x_v^j\big)^2\right)\\
    &= \frac{1}{4}\big\langle X_u, X_u\big\rangle - \frac{i}{2}\big\langle X_u, X_v\big\rangle - \frac{1}{4}\big\langle X_v, X_v\big\rangle\\
    &= \frac{1}{4}\big(E - G\big) - \frac{i}{2}F
\end{align*}

\noindent 3.
\begin{align*}
    |\varphi|^2 &:= \sum_{j=1}^{3}\big|\varphi^j\big|^2 = \sum_{j=1}^{3}\left|\frac{\partial x^j}{\partial z}\right|^2 = \sum_{j=1}^{3}\big|x_u^j - i\,x_v^j\big|^2\\
    &= \sum_{j=1}^{3}\left(\big(x_u^j\big)^2 + \big(x_v^j\big)^2\right) = \big\langle X_u, X_u\big\rangle + \big\langle X_v, X_v\big\rangle = 2E
\end{align*}
where the last equality is because $X$ is isothermal. Since $S$ is regular $\iff X_u\times X_v\neq 0 \iff \big\|X_u\times X_v\big\|^2 = EG - F^2 = E^2 > 0 \iff |\varphi|^2 = 2E\neq 0$.
\end{proof}

\begin{theorem}
\textit{Weierstrass-Enneper integral representation}. Let $D$ be a simply-connected domain in $\mathbb{C}$. Suppose a complex function $g$ is meromorphic on $D$ and a complex function $f$ is holomorphic on $D$ such that $fg^2$ is holomorphic on $D$. The holomorphic map $\varphi = (\varphi^1, \varphi^2, \varphi^3)$ given by
\[
\varphi^1 = \frac{1}{2}f\big(1 - g^2\big), \quad \varphi^2 = \frac{i}{2}f\big(1 + g^2\big), \quad\text{and}\quad \varphi^3 = fg
\]
determines a minimal surface. Conversely, given a minimal surface $S$ with isothermal parametrization $X(u, v) = (x^1, x^2, x^3)$ with domain containing the origin, there is a holomorphic map $\varphi = (\varphi^1, \varphi^2, \varphi^3)$ such that:
\[
x^j(z, \bar{z}) = c^j + 2\,\mathrm{Re}\int_\gamma\varphi^j\,dz
\]
where $\gamma$ is a path from $z_0$ to $z$ and $c^j$ are constants.
\end{theorem}

\begin{proof}
For the first part, we w.t.s.: $\varphi = \Big(\frac{1}{2}f\big(1 - g^2\big), \frac{i}{2}f\big(1 + g^2\big), fg\Big)$ determines minimal surface:
Since $fg^2$ is holomorphic, then $\varphi$ is holomorphic. Then compute:
\[
(\varphi)^2 := \big(\varphi^1\big)^2 + \big(\varphi^2\big)^2 + \big(\varphi^3\big)^2 = \frac{1}{4}f^2\big(1 - g^2\big)^2 + \frac{-1}{4}f^2\big(1 + g^2\big)^2 + f^2g^2
\]
\[
= \frac{1}{4}f^2\left(\big(1 - g^2\big)^2 - \big(1 + g^2\big)^2 + 4g^2\right) = 0.
\]
Hence, by the previous prop, we're done. For the converse part,
\[
dx^j = \frac{\partial x^j}{\partial z}\,dz + \frac{\partial x^j}{\partial\bar{z}}\,d\bar{z} := \varphi^j\,dz + \overline{\varphi}^j\,d\bar{z} = \varphi^j\,dz + \overline{\varphi^j\,dz} = 2\,\mathrm{Re}\big(\varphi^j\,dz\big).
\]
Therefore,
\[
x^j = \int_\gamma dx^j = c^j + \int_\gamma 2\,\mathrm{Re}\big(\varphi^j\,dz\big) = c^j + 2\,\mathrm{Re}\int_\gamma\varphi^j\,dz
\]
for some constant $c^j$.
\end{proof}

\noindent\textbf{Example}: $f = 1$, $g = z^2 + 2$. Then $X = \mathrm{Re}\begin{pmatrix}\int f(1 - g^2)\,dz\\ \int if(1 + g^2)\,dz\\ 2\int fg\,dz\end{pmatrix} = \mathrm{Re}\begin{pmatrix}z - \frac{z^3}{3}\\ iz + i\frac{z^3}{3}\\ z^2\end{pmatrix}$. Replace $z$ by $(u + iv)$ and then we have:
\[
X(u, v) = \begin{pmatrix}
u - \frac{u^3}{3} + uv^2\\[1mm]
-v - u^2v + \frac{v^3}{3}\\[1mm]
u^2 - v^2
\end{pmatrix}
\]
Hence, $X_u = \begin{pmatrix}1 - u^2 + v^2\\ -2uv\\ 2u\end{pmatrix}$ and $X_v = \begin{pmatrix}2uv\\ -1 - u^2 + v^2\\ -2v\end{pmatrix}$ $\Rightarrow F = 0$, $E = G = (1 + u + v)^2$, so $X$ is isothermal. Furthermore, $X_{uu} = \begin{pmatrix}-2u\\ -2v\\ 2\end{pmatrix}$ and $X_{vv} = \begin{pmatrix}2u\\ 2v\\ -2\end{pmatrix}$, so $\Delta X = 0$. Therefore, $X$ is a minimal surface. (This surface has a special name called \textit{Enneper first surface}.)

\noindent\begin{itemize}
\item \textit{Catenoid family}: Let $f = e^{-i\theta}e^z$ and $g = e^{-z}$. Then
\[
\varphi = \Big(\frac{1}{2}f\big(1 - g^2\big),\ \frac{i}{2}f\big(1 + g^2\big),\ fg\Big) = \big(e^{-i\theta}\sinh z,\ ie^{-i\theta}\cosh z,\ e^{-i\theta}\big)
\]
When $\theta = 0$, we can get the usual \textit{catenoid}. When $\theta = \frac{\pi}{2}$, we can get the \textit{helicoid}.
\end{itemize}

\subsection{Ruled Surface}

\definition{Ruled Surface}\\
A \textit{ruled surface} is swept out by a family of straight lines and locally has a parametrization
\[
X(u, v) = a(u) + v\,b(u), \quad b(u)\neq 0,
\]
where $b(u)$ is called the \textit{directrix} and $v\mapsto X(u, v)$ is a \textit{ruling} for a fixed $u$.

\begin{proposition}
Every regular ruled surface satisfies $K\le 0$.
\end{proposition}

\noindent\textbf{Example}: \textit{Helicoid}. $X(u, v) = (v\cos u, v\sin u, cu)$, $c\neq 0$ is a regular ruled surface, since for any fixed $u$, $X(u, v)$ is a straight line in the $v$ direction.

\begin{proof}
$X_v = b(u) \Rightarrow X_{vv} = 0 \Rightarrow g = 0 \Rightarrow K = \frac{-f^2}{EG - F^2}\le 0$.
\end{proof}



\begin{theorem}
\textit{Catalan's}. Every connected ruled minimal surface is contained in a plane or a helicoid (up to a rigid motion).
\end{theorem}

\begin{proof}
\textbf{Step 1}: \textit{Goal}: We want an orthonormal basis to express $X_u, X_v, X_{uu}, \cdots$, just as the Christoffel symbols.

\noindent Since $b(u)$ just gives the direction for any fixed $u$, then we may assume $|b| = 1$ and $|b'| = 1$. Consider any scalar $\lambda(u)$ and define
\[
\tilde{a}(u) := a(u) + \lambda(u)\,b(u).
\]
Since $b$ is on $S^2$, then $\langle b,T_b\rangle=0$, so we can compute:
\[
\big\langle\tilde{a}',\ b'\big\rangle = \big\langle a' + \lambda'b + \lambda b',\ b'\big\rangle = \big\langle a',\ b'\big\rangle + \lambda'\langle b, b'\rangle + \lambda\big\langle b', b'\big\rangle
\]
\[
= \big\langle a',\ b'\big\rangle + \lambda'\underbrace{\langle b, T_b\rangle}_{0} + \lambda\,|b'|^2 = \big\langle a',\ b'\big\rangle + \lambda.
\]
Hence, if we choose $\lambda(u) = -\langle a', b'\rangle$, then $\big\langle\tilde{a}', b'\big\rangle = 0$. Moreover, the two coordinate patches
\[
X(u, v) = a(u) + v\,b(u) \quad\text{and}\quad \widetilde{X}(u, v) = \tilde{a}(u) + v\,b(u) = a(u) + (v + \lambda(u))\,b(u)
\]
parametrize the same surface. Therefore, we may assume that:
\[
X(u, v) = a(u) + v\,b(u) \quad\text{is such that}\quad |b| = |b'| = 1 \quad\text{and}\quad \big\langle a', b'\big\rangle = 0,
\]
and we get an orthonormal frame $\{b,\ b',\ b\times b' =: n\}$.

\noindent\textbf{Step 2}: Now we need to express all quantities by this frame:

\noindent Since $\langle b', b'\rangle = 1$ and $\langle b', b\rangle = 0$, then
\[
\begin{cases}
0 = \big\langle b', b'\big\rangle' = 2\big\langle b'', b'\big\rangle \Rightarrow \big\langle b'', b'\big\rangle = 0\\[2mm]
0 = \big\langle b', b\big\rangle' = \big\langle b'', b\big\rangle + \big\langle b', b'\big\rangle \Rightarrow \big\langle b'', b\big\rangle = -1
\end{cases}
\]
Hence, $b'' = -b + 0\,b' + \mu\,n$ for some scalar $\mu(u)$. Moreover,
\[
n' = \big(b\times b'\big)' = b'\times b' + b\times b'' = -\mu\,b'
\]
and $a' = A(u)\,b + 0\,b' + C(u)\,n$ for some scalars $A(u), C(u)$. To conclude, we now have:
\[
b'' = -b + \mu n, \quad n' = -\mu b' \quad\text{and}\quad a' = Ab + Cn.
\]
Then we can compute the fundamental forms: $X_u = a' + v\,b' = Ab + v\,b' + Cn$, $X_v = b$, so
\[
E = A^2 + C^2 + v^2, \quad F = A \quad\text{and}\quad G = 1 \Rightarrow EG - F^2 = C^2 + v^2
\]
and moreover,
\[
X_u\times X_v = v\,b'\times b + C\,n\times b = Cb' - v\,n \Rightarrow N = \frac{Cb' - v\,n}{\sqrt{C^2 + v^2}}.
\]
$X_{uu} = (A' - v)\,b + (A - C\mu)\,b' + (C' + v\mu)\,n$, $X_{uv} = b'$, $X_{vv} = 0$, so
\[
(e, f, g) = \left(\frac{C\big(A - C\mu\big) - v\big(C' + v\mu\big)}{\sqrt{C^2 + v^2}},\ \frac{C}{\sqrt{C^2 + v^2}},\ 0\right).
\]
Hence,
\[
H = \frac{eG - 2fF + gE}{EG - F^2} = -\frac{\mu\big(C^2 + v^2\big) + vC' + AC}{\big(C^2 + v^2\big)^{\frac{3}{2}}}.
\]

\noindent\textbf{Step 3}: Since $X$ is a minimal surface, then $H = 0$ $\forall u, v$. Hence, by the formula above, we have:
\[
\mu = 0, \quad C' = 0, \quad AC = 0.
\]

\noindent\textbf{Step 4}: We now discuss $C$ in two cases.

\noindent\underline{Case 1}: $C = 0 \Rightarrow a' = Ab \Rightarrow a(u) - a(u_0) = \int_{u_0}^{u}A(s)\,b(s)\,ds$. Note that if $\mu = 0$, then $b'' = -b$, which implies that $b$ lies on a great circle of $S^2$, since $|b| = |b'| = 1$. Then $b$ is a plane curve, say $b(u)\subseteq P$ for some fixed plane $P\subseteq\mathbb{R}^3$. Hence, $a(u) - a(u_0) = \int_{u_0}^{u}A(s)\,b(s)\,ds\subseteq P \Rightarrow a(u)\subseteq a(u_0) + P$. Since the plane containing $a(u)$ is parallel to the plane containing $b(u)$, then $X = a + v\,b\subseteq a(u_0) + P$.

\noindent\underline{Case 2}: $C\neq 0 \Rightarrow A = 0 \Rightarrow a' = Cn$. Apply the same argument in case 1, $b(u)$ is a great circle of $S^2$. After a rigid motion, we may assume $b(u) = (\cos u, \sin u, 0)$. Hence, $n = (0, 0, 1)$. Therefore, $a = (0, 0, Cu)$, so
\[
X(u, v) = a(u) + v\,b(u) = (v\cos u, v\sin u, Cu)
\]
where $C$ is a constant and hence $X$ is a part of helicoid.
\end{proof}

\section{Gaussian Curvature}

\begin{theorem}
The Gaussian curvature of $p\in S$ can be computed by
\[
K = \lim_{n\to\infty}\frac{\mathrm{sgn\ area}\big(N(B_n)\big)}{B_n}
\]
where $B_1 \supseteq B_2 \supseteq \cdots \supseteq B_n \supseteq B_{n + 1} \supseteq \cdots$ are balls around $p$ with $\lim\limits_{n\to\infty}\mathrm{area}\big(B_n\big) = 0$, i.e.: $d\big(\mathrm{area}\circ N\big) = K\,d(\mathrm{area})$ where $N: S\to S^2$ Gauss map.
\end{theorem}

\begin{proof}
Let $R_n := X^{-1}\big(B_n\big)$. Then $\mathrm{area}\big(B_n\big) = \iint_{R_n}\|X_u\times X_v\|\,du\,dv$, so
\[
\mathrm{area}\big(N(B_n)\big) = \iint_{R_n}\|N_u\times N_v\|\,du\,dv
\]
\begin{lemma}
$\forall A\in M_2(\mathbb{R})$, $\forall v_1, v_2\in\mathbb{R}^2$, $\|Av_1\wedge Av_2\| = |\det(A)|\cdot\|v_1\wedge v_2\|$.
\end{lemma}

\noindent Hence, by Lemma, since $-dN_p :T_pS\to T_pS$, then $\|N_u\times N_v\| = \big|\det(-dN_p)\|\cdot\|X_u\times X_v\| := |K|\cdot\|X_u\times X_v\|$. Therefore,
\[
\lim_{n\to\infty}\frac{\mathrm{area}\big(N(B_n)\big)}{\mathrm{area}\big(B_n\big)} = \lim_{n\to\infty}\frac{\iint_{R_n}|K|\cdot\|X_u\times X_v\|\,du\,dv}{\iint_{R_n}\|X_u\times X_v\|\,du\,dv} = |K(p)|.
\]
\end{proof}

\section{Geodesics}

\subsection{Definition and Basic Characterization}

\definition{Geodesic}\\
A \textit{geodesic} on a surface $S$ is a curve $\alpha: I\to S$, s.t.: $K_g\equiv 0$ on $I$.

\begin{theorem}
\textit{Gauss}. Let $\alpha: I\to S$ p.a.l. be the shortest curve between $\alpha(a_0)$ and $\alpha(a_1)$. Then $\alpha$ is a geodesic.
\end{theorem}

\begin{proof}
Let $X$ be a coordinate patch of $S$ and let $\gamma: I\to U$ be such that $\alpha = X\circ\gamma$. Consider an arbitrary \textit{perturbing function} $h: I\to\mathbb{R}^2$ with $h(a_0) = h(a_1) = 0$, i.e.: $\gamma^t(s) := \gamma(s) + th(s): I\to U$ is such that $\gamma^t(a_0) = \gamma(a_0)$ and $\gamma^t(a_1) = \gamma(a_1)$. Then define $\alpha(t, s) := \big(X\circ\gamma^t\big)(s) = X\big(\gamma_1^t(s) + th_1(s),\ \gamma_2^t(s) + th_2(s)\big)$, and define the \textit{length function}
\[
L(t) := \int_{a_0}^{a_1}\Big\langle\frac{\partial\alpha}{\partial s}(t, s),\ \frac{\partial\alpha}{\partial s}(t, s)\Big\rangle^{\frac{1}{2}}\,ds.
\]
Compute:
\[
L'(0) = \frac{d}{dt}\int_{a_0}^{a_1}\langle\alpha_s,\ \alpha_s\rangle^{\frac{1}{2}}\,ds\Bigg|_{t = 0} = \int_{a_0}^{a_1}\frac{\langle\alpha_{ts},\ \alpha_s\rangle\big|_{t = 0}}{\Big(\langle\alpha_s,\ \alpha_s\rangle\big|_{t = 0}\Big)^{\frac{1}{2}}}\,ds \stackrel{\alpha\text{ p.a.l.}}{=\!=\!=} \int_{a_0}^{a_1}\langle\alpha_{ts},\ \alpha_s\rangle\Big|_{t = 0}\,ds.
\]
Since $\frac{\partial}{\partial s}\langle\alpha_t,\ \alpha_s\rangle\Big|_{t = 0} = \langle\alpha_{ts},\ \alpha_s\rangle\big|_{t = 0} + \langle\alpha_t,\ \alpha_{ss}\rangle\big|_{t = 0}$, then
\[
L'(0) = \int_{a_0}^{a_1}\left(\frac{\partial}{\partial s}\langle\alpha_t,\ \alpha_s\rangle\Big|_{t = 0} - \langle\alpha_t,\ \alpha_{ss}\rangle\Big|_{t = 0}\right)ds \quad (*)
\]
Since $\alpha(t, s) = X\big(\gamma_1(s) + th_1(s),\ \gamma_2(s) + th_2(s)\big)$, then
\[
\alpha_t(t, s)\Big|_{t = 0} = X_u\,h_1 + X_v\,h_2, \quad \alpha_s(t, s)\big|_{t = 0} = \alpha'(s) \quad\text{and}\quad \alpha_{ss}(t, s)\big|_{t = 0} = \alpha''(s).
\]
In particular, $\alpha_t(t, s)\big|_{t = 0}\big|_{s = a_0, a_1} = 0$, since $h(a_0) = h(a_1) = 0$. Hence, the integral of the first part of integrand in $(*)$ is 0, so we can then compute:
\[
L'(0) = -\int_{a_0}^{a_1}\langle\alpha_t,\ \alpha_{ss}\rangle\Big|_{t = 0}\,ds = -\int_{a_0}^{a_1}\big\langle h_1X_u + h_2X_v,\ \alpha''(s)\big\rangle\,ds
\]
\[
= -\int_{a_0}^{a_1}\big\langle h_1X_u + h_2X_v,\ K_nN + K_gT\big\rangle\,ds
\]
\[
= -\int_{a_0}^{a_1}\big\langle h_1X_u + h_2X_v,\ K_gT\big\rangle\,ds
\]
\[
= -K_g\int_{a_0}^{a_1}\big\langle h_1X_u + h_2X_v,\ T\big\rangle\,ds.
\]
Since the curve attains its minimum length at $t = 0$, then we have $L'(0) = 0$ for any choice of $h_1$ and $h_2$. Therefore, $K_g = 0$ on $(a_0, a_1)$.
\end{proof}

\begin{lemma}
$K_g(s) = \det\big(N(\alpha(s)),\ \alpha'(s),\ \alpha''(s)\big)$.
\end{lemma}

\begin{proof}
\[
K_g(s) = \big\langle\alpha''(s),\ T\big\rangle = \big\langle\alpha'',\ N\times\alpha'\big\rangle = \det\big(N(\alpha),\ \alpha',\ \alpha''\big).
\]
\end{proof}

\begin{corollary}
$\alpha(s)$ is a geodesic $\iff \det\big(N(\alpha),\ \alpha',\ \alpha''\big) = 0$.
\end{corollary}

\noindent\textbf{Example}:
\begin{itemize}
\item $S = S^2$. Every two points on $S^2$ can be joined by a geodesic. Let $p, q\in S^2$. If $p = \pm q$, then $\exists A\in O(3)$, s.t.: $Ap = e_1$ and then $Aq = \pm e_1$, so $p, q$ can be transformed onto the equator by a rigid motion. Otherwise, if $p\neq\pm q$, then define $r := q - \langle q, p\rangle p$. Since
\[
\|r\|^2 = \|q\|^2 - 2\langle q, p\rangle^2 + \langle q, p\rangle^2\|p\|^2 = 1 - \langle q, p\rangle^2 > 0
\]
then we can define $w := r/\|r\|$. Then $\langle w, p\rangle = 0$, which means that $\{p, w, p\times w\}$ is an orthonormal basis for $\mathbb{R}^3$. Hence, $\exists A\in O(3)$, s.t.: $A$ is a linear operator that maps $\{p, w, p\times w\}$ to $\{e_1, e_2, e_3\}$. Since
\[
\langle q, p\rangle^2 + \langle q, w\rangle^2 = \langle q, p\rangle^2 + \left(\frac{\langle q, q\rangle - \langle q, p\rangle^2}{\sqrt{1 - \langle q, p\rangle^2}}\right)^2 = 1
\]
then $Aq = (\cos\theta, \sin\theta, 0)$. Since $Ap = e_1$, then $p, q$ can be transformed onto the equator by a rigid motion. Hence, we may assume that $p$ and $q$ are joined by the equator $\alpha(s) = (\cos(s), \sin(s), 0)$. Then $\alpha'' = -\alpha$ and $N(\alpha(s)) = \alpha(s)$. Therefore,
\[
\det\big(N\alpha,\ \alpha',\ \alpha''\big) = 0 \iff \alpha\text{ is a geodesic.}
\]
In fact, the great circles are all geodesics.
\item A curve $\alpha(t) = \big(\cos(at),\ \sin(at),\ q(t)\big)$ on cylinder is a geodesic iff $q''(t) = 0$.

\noindent\textit{Proof}: Let $X(u, v) = (\cos u, \sin u, v)$ be a parametrization of cylinder. Then $X_u = (-\sin u, \cos u, 0)$ and $X_v = (0, 0, 1)$. Hence,
\[
N = \frac{X_u\times X_v}{\|X_u\times X_v\|} = (\cos u, \sin u, 0)
\]
so $N(\alpha(t)) = (\cos(at), \sin(at), 0)$. Since $\alpha'(t) = (-a\sin(at), a\cos(at), q'(t))$ and $\alpha''(t) = (-a^2\cos(at), -a^2\sin(at), q''(t))$, then
\[
\det\big(N(\alpha(t)),\ \alpha'(t),\ \alpha''(t)\big) = \begin{vmatrix}
\cos(at) & -a\sin(at) & -a^2\cos(at)\\
\sin(at) & a\cos(at) & -a^2\sin(at)\\
0 & q'(t) & q''(t)
\end{vmatrix}
\]
\[
= -q'(t)\begin{vmatrix}\cos(at) & -a^2\cos(at)\\ \sin(at) & -a^2\sin(at)\end{vmatrix} + q''(t)\begin{vmatrix}\cos(at) & -a\sin(at)\\ \sin(at) & a\cos(at)\end{vmatrix}
\]
\[
= aq''(t)
\]
Therefore, $\alpha$ is a geodesic $\iff \det\big(N\alpha,\ \alpha',\ \alpha''\big) = 0 \iff q''(t) = 0$. $\square$

\subsection{Christoffel Symbols}

\definition{Christoffel Symbols}\\
Let $X$ be a coordinate patch of $S$ with redefining $X_1 := X_u$ and $X_2 := X_v$. The \textit{Christoffel symbols} for $X$ is a collection $\big\{\Gamma_{ij}^k: U\to\mathbb{R} \mid i, j, k\in\{1, 2\}\big\}$ of functions satisfying
\[
X_{uu} = \Gamma_{11}^1X_1 + \Gamma_{11}^2X_2 + eN
\]
\[
X_{uv} = \Gamma_{12}^1X_1 + \Gamma_{12}^2X_2 + fN
\]
\[
X_{vu} = \Gamma_{21}^1X_1 + \Gamma_{21}^2X_2 + fN
\]
\[
X_{vv} = \Gamma_{22}^1X_1 + \Gamma_{22}^2X_2 + gN.
\]

\noindent\textbf{Remark}:
\begin{itemize}
\item The coefficients of $N$ in the above formulas are given by: $e = \langle X_{uu}, N\rangle$, $f = \langle X_{uv}, N\rangle = \langle X_{vu}, N\rangle$ and $g = \langle X_{vv}, N\rangle$.
\item Note that $\Gamma_{12}^1 = \Gamma_{21}^1$ and $\Gamma_{12}^2 = \Gamma_{21}^2$. Hence, we often ignore $\Gamma_{21}^1$ and $\Gamma_{21}^2$ in the Christoffel symbols.
\end{itemize}

\begin{theorem}
\textit{Geodesic Equations}. $\alpha(s) = \big(X\circ\beta\big)(s) = X\big(u(s), v(s)\big)$ is a geodesic iff the following two equations (\textit{geodesic equations}) hold:
\[
\begin{cases}
u'' + \Gamma_{11}^1\,(u')^2 + 2\Gamma_{12}^1\,u'v' + \Gamma_{22}^1\,(v')^2 = 0\\[2mm]
v'' + \Gamma_{11}^2\,(u')^2 + 2\Gamma_{12}^2\,u'v' + \Gamma_{22}^2\,(v')^2 = 0
\end{cases}
\]
\end{theorem}

\begin{proof}
$\alpha' = X_u\,u' + X_v\,v'$. Then
\[
\alpha'' = \big(X_{uu}\,u' + X_{uv}\,v'\big)\,u' + X_u\,u'' + \big(X_{vu}\,u' + X_{vv}\,v'\big)\,v' + X_v\,v''
\]
\[
= X_{uu}\,(u')^2 + X_u\,u'' + 2X_{uv}\,u'v' + X_v\,v'' + X_{vv}\,(v')^2
\]
\[
= \left(\Gamma_{11}^1\,(u')^2 + 2\Gamma_{12}^1\,u'v' + \Gamma_{22}^1\,(v')^2 + u''\right)X_u + *N +
\]
\[
\left(\Gamma_{11}^2\,(u')^2 + 2\Gamma_{12}^2\,u'v' + \Gamma_{22}^2\,(v')^2 + v''\right)X_v.
\]
Hence, $\alpha$ is a geodesic $\iff K_g = 0 \iff \alpha'' = K_nN \iff \alpha''\parallel N \iff$ the coefficients of both $X_u$ and $X_v$ are 0.
\end{proof}

\begin{theorem}
\textit{Gauss Equation}. Let $I = \begin{pmatrix}E & F\\ F & G\end{pmatrix}$ be the f.f.f. and $u_1, u_2 = (u, v)$. Then
\[
\Gamma_{ij}^k = \frac{1}{2}\sum_{\ell=1}^{2}\big(I^{-1}\big)_{k\ell}\left(\frac{\partial I_{i\ell }}{\partial u_j} + \frac{\partial I_{\ell j}}{\partial u_i} - \frac{\partial I_{ij}}{\partial u_\ell}\right)
\]
\end{theorem}

\begin{proof}
Compute:
\[
\frac{\partial I_{i\ell}}{\partial u_j} = \frac{\partial}{\partial u_j}\big\langle X_i, X_\ell\big\rangle = \big\langle X_{ij},\ X_\ell\big\rangle + \big\langle X_i,\ X_{\ell j}\big\rangle
\]
\[
\frac{\partial I_{\ell j}}{\partial u_i} = \frac{\partial}{\partial u_i}\big\langle X_\ell, X_j\big\rangle = \big\langle X_{\ell i},\ X_j\big\rangle + \big\langle X_\ell,\ X_{ji}\big\rangle
\]
\[
\frac{\partial I_{ij}}{\partial u_\ell} = \frac{\partial}{\partial u_\ell}\big\langle X_i, X_j\big\rangle = \big\langle X_{i\ell},\ X_j\big\rangle + \big\langle X_i,\ X_{j\ell}\big\rangle
\]
Hence,
\[
\frac{1}{2}\left(\frac{\partial I_{i\ell}}{\partial u_j} + \frac{\partial I_{\ell j}}{\partial u_i} - \frac{\partial I_{ij}}{\partial u_\ell}\right) = \big\langle X_{ij},\ X_\ell\big\rangle \quad (*)
\]
Since $X_{ij} = \Gamma_{ij}^1X_1 + \Gamma_{ij}^2X_2 + \#N$, then
\[
\begin{cases}
\big\langle X_{ij},\ X_1\big\rangle = E\,\Gamma_{ij}^1 + F\,\Gamma_{ij}^2\\[2mm]
\big\langle X_{ij},\ X_2\big\rangle = F\,\Gamma_{ij}^1 + G\,\Gamma_{ij}^2
\end{cases}
\]
Therefore, by $(*)$, we're done from:
\[
I\begin{pmatrix}\Gamma_{ij}^1\\ \Gamma_{ij}^2\end{pmatrix} = \begin{pmatrix}\langle X_{ij},\ X_1\rangle\\ \langle X_{ij},\ X_2\rangle\end{pmatrix} \iff \begin{pmatrix}\Gamma_{ij}^1\\ \Gamma_{ij}^2\end{pmatrix} = I^{-1}\begin{pmatrix}\langle X_{ij},\ X_1\rangle\\ \langle X_{ij},\ X_2\rangle\end{pmatrix}
\]
\end{proof}

\noindent\textbf{Example}: Consider a surface of revolution $X(u, v) = \big(f(v)\cos u,\ f(v)\sin u,\ g(v)\big)$. Then $X_u = \big(-f(v)\sin u,\ f(v)\cos u,\ 0\big)$, $X_v = \big(f'(v)\cos u,\ f'(v)\sin u,\ g'(v)\big)$, so $E = f^2$, $F = 0$, $G = (f')^2 + (g')^2$. Hence, the f.f.f. is:
\[
I = \begin{pmatrix}f^2 & 0\\ 0 & (f')^2 + (g')^2\end{pmatrix}
\]
which gives: $\frac{\partial I_{11}}{\partial u_1} = 0$, $\frac{\partial I_{11}}{\partial u_2} = 2ff'$, $\frac{\partial I_{12}}{\partial u_1} = 0$, $\frac{\partial I_{12}}{\partial u_2} = 0$, $\frac{\partial I_{22}}{\partial u_1} = 0$, $\frac{\partial I_{22}}{\partial u_2} = 2f'f'' + 2g'g''$ and $I^{-1} = \frac{1}{f^2\big((f')^2 + (g')^2\big)}\begin{pmatrix}(f')^2 + (g')^2 & 0\\ 0 & f^2\end{pmatrix}$. Then by the Gauss equation, we get
\[
\Gamma_{11}^1 = 0,  \quad \Gamma_{12}^1 = \frac{f'}{f}, \quad \Gamma_{22}^1 = 0, \quad \Gamma_{11}^2 = \frac{-ff'}{(f')^2 + (g')^2}, \quad \Gamma_{12}^2 = 0, \quad \Gamma_{22}^2 = \frac{f'f'' + g'g''}{(f')^2 + (g')^2}.
\]
Then take them into the geodesic equation:
\[
\begin{cases}
u'' + 2\big(\frac{f'}{f}\big)\,u'v' = 0\\[2mm]
v'' - \frac{ff'}{(f')^2 + (g')^2}\,(u')^2 + \frac{f'f'' + g'g''}{(f')^2 + (g')^2}\,(v')^2 = 0
\end{cases}
\]

\noindent\begin{itemize}
\item When $v = \mathrm{constant}$ and $u = u(s)\neq\mathrm{constant}$, i.e.: the curve $\alpha(s) = X(u(s), v_0)$, then $\alpha$ is geodesic iff $\alpha$ satisfies
\[
u'' = 0 \quad\text{and}\quad \frac{ff'}{(f')^2 + (g')^2}\,(u')^2 = 0
\]
Since $u\neq\mathrm{constant}$, then the above conditions is equivalent to: $u(s) = as + b$ $(a\neq 0)$ and $f' = 0$.
\item When $u = \mathrm{constant}$ and $v = v(s)$, i.e.: the curve $\alpha(s) = X(u_0, v(s))$, then $\alpha$ is geodesic iff $\alpha$ satisfies
\[
v''\big((f')^2 + (g')^2\big) + (v')^2\big(f'f'' + g'g''\big) = 0
\]
In particular, every \textit{meridian curve} is geodesic, i.e.: the curve parametrized of constant speed is geodesic.

\noindent\textit{Proof}: $\alpha(s) = X(u_0, v(s))$. Since $\alpha$ is a meridian curve, i.e. $\alpha$ is parametrized of constant speed, then constant $= |\alpha'| = (f')^2 + (g')^2 $, so $ f'f'' + g'g'' = 0$. Moreover, $v(s) = s \Rightarrow v'' = 0$. Therefore,
\[
v''\left((f')^2 + (g')^2\right) + (v')^2\big(f'f'' + g'g''\big) = 0\cdot\big((f')^2 + (g')^2\big) + (v')^2\cdot 0 = 0 \quad\square
\]
\item If $\alpha(s) = X\big(u(s), v(s)\big)$ has constant speed, then $\big\langle\alpha', \alpha'\big\rangle = 0 \Rightarrow \big\langle\alpha'', \alpha'\big\rangle = 0$. On the other hand, by the proof of the geodesics equations thm,
\[
\alpha'' = \left(u'' + \Gamma_{11}^1\,(u')^2 + 2\Gamma_{12}^1\,u'v' + \Gamma_{22}^1\,(v')^2\right)X_u + *N
\]
\[
+ \left(v'' + \Gamma_{11}^2\,(u')^2 + 2\Gamma_{12}^2\,u'v' + \Gamma_{22}^2\,(v')^2\right)X_v
\]
Hence, plug $\Gamma_{11}^1 = \Gamma_{22}^1 = \Gamma_{12}^2 = 0$ and the other $\Gamma_{ij}^k$'s in:
\[
\alpha'' = \left(u'' + 2\big(\tfrac{f'}{f}\big)\,u'v'\right)\,X_u + *N + \left(v'' + \frac{-ff'}{(f')^2 + (g')^2}\,(u')^2 + \frac{f'f'' + g'g''}{(f')^2 + (g')^2}\,(v')^2\right)X_v
\]
Since $E = f^2$, $F = 0$ and $G = (f')^2 + (g')^2$, then
\[
0 = \big\langle\alpha'', \alpha'\big\rangle = \left(u'' + 2\big(\tfrac{f'}{f}\big)u'v'\right)f^2\,u' + \left(v'' + \frac{-ff'}{(f')^2 + (g')^2}\,(u')^2 + \frac{f'f'' + g'g''}{(f')^2 + (g')^2}\,(v')^2\right)\big((f')^2 + (g')^2\big)\,v'
\]
\[
= u''f^2\,u' + 2ff'(u')^2\,v' + v''\big((f')^2 + (g')^2\big)v' - ff'\,(u')^2\,v' + \big(f'f'' + g'g''\big)(v')^3
\]
\[
= u''f^2\,u' + ff'\,(u')^2\,v' + \big((f')^2 + (g')^2\big)v'v'' + \big(f'f'' + g'g''\big)(v')^3
\]
If we pause for a moment in the first row, then pluging $u'' + 2\big(\tfrac{f'}{f}\big)u'v' = 0$ in, we can get:
\[
\left(v'' - \frac{ff'}{(f')^2 + (g')^2}\,(u')^2 + \frac{f'f'' + g'g''}{(f')^2 + (g')^2}\,(v')^2\right)\big((f')^2 + (g')^2\big)\,v' = 0
\]
For $v\neq\mathrm{constant}$, $v'' - \frac{ff'}{(f')^2 + (g')^2}\,(u')^2 + \frac{f'f'' + g'g''}{(f')^2 + (g')^2}\,(v')^2 = 0$. This shows that: if $\alpha$ has constant speed and $v\neq\mathrm{constant}$, then satisfying the first geodesic equation gives $\alpha$ a geodesic.

\noindent Furthermore, note that
\[
u'' + 2\big(\tfrac{f'}{f}\big)\,u'v' = 0 \iff \frac{u''}{u'} + 2\big(\tfrac{f'(v)}{f}\big)\,v' = 0 \iff \big(\log(u')\big)' + 2\big(\log(f)\big)' = 0
\]
\[
\iff \Big(\log\big(u'f^2\big)\Big)' = 0 \iff u'f^2 = \mathrm{constant}
\]
and that if we write the angle between $\alpha'$ and $X_u$ as $\theta$, then
\[
\alpha'f\cos\theta = \|\alpha'\|\,\|X_u\|\cos\theta = \big\langle\alpha', X_u\big\rangle = \big\langle X_u\,u' + X_v\,v', X_u\big\rangle = u'E = u'f^2.
\]
Hence, since $\alpha$ has constant speed, then
\[
u'' + 2\big(\tfrac{f'}{f}\big)\,u'v' = 0 \iff f\cos\theta = \mathrm{constant} \quad\big(\textit{Clairaut relation}\big).
\]
\end{itemize}

\noindent In particular, we now can describe the following specific surfaces of revolution: torus ($f(v)=a+r\cos v, g(v)= r\sin v$), hyperbolic ($f(v)= \sqrt{1+v^2} ,g(v)=v$), parabolic ($f(v)= v ,g(v)=\frac{v^2}{2}$) and catenoid ($f(v)=\cosh v ,g(v)=v$):
\begin{itemize}
    \item On \textbf{torus}, every meridian (the generating circle) is a geodesic, and the inner and outer equators are geodesics. (WARN that the general parallel circles instead of equators are \textit{not} geodesics, since $\alpha''$ is horizontal but the normal $N$ is not, i.e.: $\alpha'' \nparallel N$.) Furthermore, for the general curve with constant speed $\alpha(t)= \big((a+r\cos v(t))\cos u(t), (a+r\cos v(t)) \sin u(t), r\sin v(t)\big)$, it is a geodesic if the Clairaut relation, $f\cos \theta=$ constant $=:\lambda \in [-(a+r),a+r]$, is satisfied:
    \begin{itemize}
        \item When $\lambda=0$, then $\theta=\pi/2 \Leftrightarrow \alpha' \perp X_u \Leftrightarrow u'=0 \Leftrightarrow u=$ constant $\Leftrightarrow \alpha$ is a meridian.
        \item When $|\lambda|=a+r$, then $v=0$ and $\theta=0 \Leftrightarrow v=$ constant, which is the case of outer equator. Similarly, when $|\lambda|=a-r$, then it is the case of inner equator.
    \end{itemize}
    \noindent For $0<|\lambda|<a+r$ and $|\lambda|\neq a-r$, note that $|f| = \frac{|\lambda|}{|\cos \theta|} \geq |\lambda|$. Hence, if we look down at this torus, then the area $x^2+y^2 <|\lambda|^2$ is a "forbidden zone" for $\alpha$. Therefore, this gives us other cases for $|\lambda|$:
    \begin{itemize}
        \item When $0<|\lambda|<a-r$, then the "forbidden zone" does \textit{not} intersect torus, which means that $\alpha$ can form a spiral around torus. Furthermore, recall the Gaussian curvature of torus
        \[
        K=\frac{\cos v}{r(a+r\cos v)} \begin{cases}
            >0 \text{ ,when } -\frac{\pi}{2}<v<\frac{\pi}{2}\\
            =0 \text{ ,when } v=\frac{\pi}{2} \text{ or } \frac{3\pi}{2}\\
            <0 \text{ ,when } \frac{\pi}{2}<v <\frac{3\pi}{2}
        \end{cases}
        \]
        Imagine the outer part, say $O$, and inner part, say $I$, between the top and bottom circles of torus. Then, geometrically, this Gaussian curvature gives: $\alpha$ bends relatively slowly in $O$ and relatively fast in $I$.
        \item When $a-r<|\lambda|<a+r$, then a "forbidden zone" appears on torus. This means that $\alpha$ can only "oscillate" within a band-like area outside the "forbidden zone" and smoothly turns back when touching the circles $\{ (x,y,g(v)) \mid x^2+y^2= |\lambda|^2, f(v)=|\lambda|\}$.
    \end{itemize}
    
    \item On \textbf{hyperbolic}, since for torus, the particular cases will appear when analyzing $\lambda$, then we directly look at the general curve with constant speed $\alpha(t)= \big(\sqrt{1+v(t)^2}\cos u(t), \sqrt{1+v(t)^2} \sin u(t), v \big)$. It is a geodesic if the Clairant relation, $f\cos \theta=: \lambda \in \mathbb{R}$, is satisfied:
    \begin{itemize}
        \item When $\lambda=0$, then similarly as in torus with the case of $\lambda=0$, this gives the meridian.
        \item When $|\lambda|=1$, $v=0$, so $\alpha$ is the equator.
        \item When $0<|\lambda|<1$, as in torus, the "forbidden zone" doesn't intersects hyperbolic. Since $g(v)=v \to \pm\infty$ as $v \to \pm\infty$, then we can express $\alpha$ in the method of ruled surface, i.e.: $\alpha \in \{\alpha_{\beta,+}, \alpha_{\beta,-} :\mathbb{R} \to S\}_{\beta\in [0,2\pi]}$, where
        \[
        \alpha_{\beta,\pm} (t):= \big(\cos \beta- t\sin \beta, \sin \beta +t\cos \beta, \pm t\big)
        \]
        which is easy to check that this family of geodesics are lying on the hyperbolic, i.e.: every $\alpha_{\beta,\pm}$ satisfies $x^2+y^2-z^2=1$.
        \item When $|\lambda|>1$, then a "forbidden zone" appears on hyperbolic, stay around equator like a scarf. This means that $\alpha$ can only "oscillate" above or below the "forbidden zone" and smoothly turns back when touching the circles $\{ (x,y,v) \mid x^2+y^2=|\lambda|^2, \sqrt{1+v^2}=|\lambda|\}$. Furthermore, since the closer hyperbolic go to the middle, the more it bend, then the closer $\alpha$ go to middle, the densor it will become.
    \end{itemize}

    \item On \textbf{parabolic}, for the general curve with constant speed $\alpha(t)= (v(t)\cos u(t), v(t)\sin u(t), \frac{v^2}{2})$, it is a geodesic if the Clairant relation, $f\cos \theta=:\lambda \in \mathbb{R}$, is satisfied:
    \begin{itemize}
        \item When $\lambda=0$, then $\alpha$ is a meridian.
        \item When $\lambda \neq 0$, then a "forbidden zone" appears on the parabolic. This means that $\alpha$ can only appear above the circle $\{ (x,y,\frac{\lambda^2}{2}) \mid x^2+y^2 = \lambda^2\}$. Furthermore, since the lower parabolic goes, the more it bend, then the lower $\alpha$ go, the denser it will become.
    \end{itemize}

    \item On \textbf{catenoid}, for the general curve with constant speed $\alpha(t)= (\cosh v(t) \cos u(t), \cosh v(t)\sin u(t), v(t))$, it is a geodesic if the Clairant relation, $f\cos \theta=: \lambda \in \mathbb{R}$, is satisfied:
    \begin{itemize}
        \item When $\lambda=0$, then $\alpha$ is a meridian.
        \item When $|\lambda|=1$, then $\alpha$ is the equator.
        \item When $0<|\lambda|<1$, then $\alpha$ can intersect the equator.
        \item When $|\lambda|>1$, then a "forbidden zone" appears on hyperbolic, stay around equator like a scarf. This means that $\alpha$ can only appear above or below the the circles $\{ (x,y,v) \mid x^2+y^2=|\lambda|^2, |v|\geq \text{arccosh} |\lambda|\}$. Furthermore, since the closer hyperbolic go to the middle, the more it bend, then the closer $\alpha$ go to middle, the densor it will become.
    \end{itemize}
    \noindent Furthermore, since \textbf{helicoid} can be transformed via isometric deformation by catenoid and isometry preserves geodesic, then the description of geodesic on helicoid is the same as catenoid.
\end{itemize}

\section{Fundamental Theorem of Surface}

\noindent\textbf{Notation}: $\Phi = \big(X_u,\ X_v,\ N\big): S\to GL\big(\mathbb{R}^3\big)$, $-dN = \big(a_{ij}\big)$,
\[
\Lambda := \begin{pmatrix}
\Gamma_{11}^1 & \Gamma_{12}^1 & -a_{11}\\
\Gamma_{11}^2 & \Gamma_{12}^2 & -a_{21}\\
e & f & 0
\end{pmatrix}
\quad\text{and}\quad
\Theta := \begin{pmatrix}
\Gamma_{21}^1 & \Gamma_{22}^1 & -a_{12}\\
\Gamma_{21}^2 & \Gamma_{22}^2 & -a_{22}\\
f & g & 0
\end{pmatrix}.
\]
Then we can check that
\[
\Phi_u = \big(X_{uu},\ X_{vu},\ N_u\big)
\]
\[
= \Big(\Gamma_{11}^1X_u + \Gamma_{11}^2X_v + eN,\ \Gamma_{12}^1X_u + \Gamma_{12}^2X_v + fN,\ -dN\big(X_u\big)\Big)
\]
\[
= \Big(\Gamma_{11}^1X_u + \Gamma_{11}^2X_v + eN,\ \Gamma_{12}^1X_u + \Gamma_{12}^2X_v + fN,\ -a_{11}X_u - a_{21}X_v\Big)
\]
\[
= \Phi\Lambda
\]
and similarly that $\Phi_v = \Phi\Theta$.

\begin{theorem}
\textit{Gauss-Weingarten Equations}. The matrices $\Lambda, \Theta$ satisfy the following:
\[
\frac{\partial\Lambda}{\partial v} - \frac{\partial\Theta}{\partial u} = \Lambda\Theta - \Theta\Lambda
\]
which is the \textit{compatibility condition}.
\end{theorem}

\begin{proof}
\begin{align*}
    \Phi_{uv} &= \big(\Phi\Lambda\big)_v = \Phi_v\Lambda + \Phi\Lambda_v = \Phi\Theta\Lambda + \Phi\Lambda_v = \Phi\big(\Theta\Lambda + \Lambda_v\big)\\
    \Phi_{vu} &= \big(\Phi\Theta\big)_u = \Phi_u\Theta + \Phi\Theta_u = \Phi\Lambda\Theta + \Phi\Theta_u = \Phi\big(\Lambda\Theta + \Theta_u\big)
\end{align*}
Hence, $\Theta\Lambda + \Lambda_v = \Lambda\Theta + \Theta_u \iff \Lambda_v - \Theta_u = \Lambda\Theta - \Theta\Lambda$.
\end{proof}
