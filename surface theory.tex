\chapter{Surface Theory}

\section{Local Theory of Surfaces}

\subsection{Regular Surface}

\definition{Parametrized Surface}\\
A \textit{parametrized surface} is a function $X: U\subseteq\mathbb{R}^2\to\mathbb{R}^3$ which is differentiable. Furthermore, the image of $X$ is called the \textit{trace} of $X$.

\noindent\textbf{Example}: $X: [0, 2\pi)\times[0, \pi]\to\mathbb{R}^3$ given by
\[
(\theta, \varphi)\longmapsto\begin{pmatrix} r\cos\theta\cdot\sin\varphi\\ r\sin\theta\cdot\sin\varphi\\ r\cos\varphi \end{pmatrix}
\]
gives a sphere $S^2$. But this raises problems:

\noindent\begin{itemize}
\item $\forall\theta\in[0, 2\pi)$, $X(\theta, 0) = (0, 0, r)$, which means that when $\varphi = 0$, $X$ is not bijective.
\item Since $\dfrac{\partial X}{\partial\theta}\bigg|_{(0,0)} = (0,0,0)$, then $\dfrac{\partial X}{\partial\theta}$ and $\dfrac{\partial X}{\partial\varphi}$ are linearly dependent at $(0,0)$, so the Jacobian matrix has rank 1. This means that at $(0,0)$, we can not compute the ``normal vector''.
\end{itemize}

\noindent To fix these problems, we use semi circles to cover $S^2$, since each semi circle can be parametrized by a (unit) open ball in $\mathbb{R}^2$.

\definition{Regular Surface}\\
A subset $S\subseteq\mathbb{R}^3$ is called a \textit{regular surface} if $\forall p\in S$, $\exists$ a neighborhood $V\subseteq\mathbb{R}^3$ of $p$ ($V$ can be seen as an open ball centered at $p$ and radius of some $\varepsilon > 0$), and a map $X: U\to V\cap S$ for some (open connected) $U\subseteq\mathbb{R}^2$, such that the followings hold:

\begin{enumerate}
\item $X$ is differentiable, i.e.: $\forall (u, v)\in U$, $X(u, v) = (x(u,v), y(u,v), z(u,v))$ has partial derivatives of all orders.
\item $X$ is a homeomorphism from $U$ onto $V\cap S$.
\item $X$ is \textit{regular}, i.e.: $\forall q\in U$,
\[
dX_q = \begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q)
\]
is of rank 2.
\end{enumerate}

\noindent\textbf{Example}:

\noindent\begin{itemize}
\item Let $S^2 = \{(x, y, z)\in\mathbb{R}^3 \mid x^2 + y^2 + z^2 = 1\}$. We use 6 coordinate patches to cover $S^2$: Let $U := \{(u, v)\in\mathbb{R}^2 \mid u^2 + v^2 < 1\}$. Consider a map $X_1: U\to V\cap S^2$ given by
\[
(u, v)\longmapsto\left(u, v, \sqrt{1 - u^2 - v^2}\right)
\]
Then \underline{check}:

\begin{enumerate}
\item $u$, $v$, $\sqrt{1-u^2-v^2}$ have infinite partial derivatives.
\item $X_1$ is a bijection, with $X_1^{-1}: (x, y, z)\mapsto(x, y)$ continuous.
\item $\forall (u, v)\in U$, compute
\[
(dX_1)_q = \begin{pmatrix}1 & 0\\ 0 & 1\\ * & \#\end{pmatrix}(q)
\]
where $\begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}$ is of rank 2, so $(dX_1)_q$ is of rank 2.
\end{enumerate}

\noindent Similarly, we can define
\[
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_2}\begin{pmatrix}u\\ v\\ -\sqrt{1-u^2-v^2}\end{pmatrix},\quad
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_3}\begin{pmatrix}u\\ \sqrt{1-u^2-v^2}\\ v\end{pmatrix},\quad
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_4}\begin{pmatrix}u\\ -\sqrt{1-u^2-v^2}\\ v\end{pmatrix}
\]
\[
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_5}\begin{pmatrix}\sqrt{1-u^2-v^2}\\ u\\ v\end{pmatrix},\quad\text{and}\quad
\begin{pmatrix}u\\ v\end{pmatrix}\xmapsto{X_6}\begin{pmatrix}-\sqrt{1-u^2-v^2}\\ u\\ v\end{pmatrix}
\]
Then $\forall p\in S^2$, $p$ must lie in (at least) one $V_i\cap S^2$, so we can take $\{X_i: U\to V_i\cap S^2 \mid i = 1, 2, \cdots, 6\}$ to conclude that $S^2$ is a regular surface.
\end{itemize}

\noindent\begin{itemize}
\item Let $f: U\subseteq\mathbb{R}^2\to\mathbb{R}$ be a smooth function. Consider the \textit{graph} of $f$
\[
S := \mathrm{graph}(f) := \left\{\begin{pmatrix}x\\ y\\ z\end{pmatrix}\in\mathbb{R}^3 \ \Bigg|\ \begin{pmatrix}x\\ y\end{pmatrix}\in U,\ z = f(x, y)\right\}
\]
To prove that $S$ is a regular surface, in fact, we only need one coordinate patch $X: U\to S$ given by $(u, v)\mapsto(u, v, f(u, v))$. \underline{Check}:

\begin{enumerate}
\item $u$, $v$, $f(u, v)$ are all differentiable (smooth).
\item $X$ is bijective, with $X^{-1}: (x, y, z)\mapsto(x, y)$ continuous.
\item $dX_q = \begin{pmatrix}1 & 0\\ 0 & 1\\ * & \#\end{pmatrix}(q)$ is of rank 2, $\forall q\in U$.
\end{enumerate}

\noindent Therefore, $S = \mathrm{graph}(f)$ is a regular surface. Furthermore, we may ask, does the converse hold?

\begin{proposition}{Regular Surfaces are Locally Graphs}\\
Any regular surface $S$ is locally a graph, or equivalently, in combination with the above example, we conclude that a surface $S\subseteq\mathbb{R}^3$ is regular iff $S$ is locally a graph of some smooth function.
\end{proposition}

\begin{proof}
It suffices to prove the first statement: $\forall p\in S$, let $X: U\xrightarrow{\cong} V\cap S$ satisfy the three conditions, with $p = X(q)$. By the regularity of $X$,
\[
dX_q = \begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q)
\]
is of rank 2, so at least one of the three matrices below has nonzero determinant:
\[
\begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\end{pmatrix}(q),\quad
\begin{pmatrix}\dfrac{\partial x}{\partial u} & \dfrac{\partial x}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q),\quad\text{and}\quad
\begin{pmatrix}\dfrac{\partial y}{\partial u} & \dfrac{\partial y}{\partial v}\\[2mm] \dfrac{\partial z}{\partial u} & \dfrac{\partial z}{\partial v}\end{pmatrix}(q)
\]
WLOG, assume the first one has nonzero determinant, so by the Inverse Function Thm, the function $f{(u,v)} := (x(u,v), y(u,v))$, which is smooth since $X$ and $(x, y, z)\mapsto(x, y)$ are smooth, has a differentiable inverse $g = f^{-1}: (x, y)\mapsto(u(x,y), v(x,y))$ onto some neighborhood of $q$. That is,
\[
X\circ g: (x, y)\longmapsto\left(x, y, z\big(u(x,y), v(x,y)\big)\right)
\]
where $z: \mathbb{R}^2\to\mathbb{R}$ given by $(x, y)\mapsto z\big(u(x,y), v(x,y)\big)$ is the desired graph function, which is smooth, on a neighborhood of $p = X(q)$. 
\end{proof}

\item Let $f: \mathbb{R}^3\to\mathbb{R}$ be a smooth map and let $a\in\mathbb{R}$ be such that $\forall p\in f^{-1}(a)$ with $df_p\neq 0$, i.e.: $a$ is a \textit{regular value} of $f$. Then $f^{-1}(a)\subseteq\mathbb{R}^3$ is a regular surface.


\begin{proof}
$\forall p\in f^{-1}(a)$, assume WLOG that $\dfrac{\partial f}{\partial z}(p)\neq 0$. Consider $F := f - c_a$ where $c_a$ is the constant map. Then $\forall p\in f^{-1}(a)$, $F(p) = f(p) - a = 0$ and $\dfrac{\partial F}{\partial z}(p) = \dfrac{\partial f}{\partial z}(p)\neq 0$. Hence, we can apply the Implicit Function Thm, $\exists\ g: U\subseteq\mathbb{R}^2\to\mathbb{R}$ smooth, s.t.: $F(u, v, g(u,v)) = 0$. Consider
\[
W := \{(u, v, g(u,v)) \mid (u, v)\in U\}
\]
containing $p$ clearly. Since $F(W) = \{0\}$, then $p\in W\subseteq F^{-1}(0)$. Therefore, $\exists$ a neighborhood $V$ of $p$, s.t.:
\[
V\cap f^{-1}(a) = V\cap F^{-1}(0) = W = \{(x, y, z)\ \big|\ (x, y)\in U,\ z = g(x, y)\}
\]
i.e.: $f^{-1}(a)$ is locally a graph of smooth function. 
\end{proof}

\item Consider $\frac{x^2}{a^2}+ \frac{y^2}{b^2}+ \frac{z^2}{c^2} =1$. Let $F= \frac{x^2}{a^2}+ \frac{y^2}{b^2}+ \frac{z^2}{c^2} -1$, which is clearly differentiable. $\forall w= (x,y,z)\in F^{-1} (0)$, since WLOG, $x \neq 0$, then $dF_w \neq 0$, so $0$ is a regular value of $F$. Hence, by above, $F^{-1} (0) = \{(x,y,z) \mid \frac{x^2}{a^2}+ \frac{y^2}{b^2}+ \frac{z^2}{c^2} =1\} \subseteq \mathbb{R}^3$ is a regular surface.
\end{itemize}

\subsection{Change of Coordinates}

\end{center}
That is, define $X: \mathbb{R}^2 \to S^2 \subseteq \mathbb{R}^3$ by $X(u,v) :=$ the intersection of the straight line joining $(u,v,0)$ and $(0,0,2)$. This gives a coordinate patch of $S^2 \setminus \{ (0,0,2)\}$. This gives a way to cover $S^1$, different from the first example of regular surface. Hence, there are many ways to get coordinate patches for a fixed surface.

\begin{proposition}{Change of Coordinates}\\
Let $p\in S$ be in the image of two parametrizations $X: U\to S$, $Y: V\to S$, i.e.: $p\in W = X(U)\cap Y(V)$. Then
\[
\psi := X^{-1}\circ Y: Y^{-1}(W)\longrightarrow X^{-1}(W)
\]
is a diffeomorphism.
\end{proposition}

\begin{proof}
Since $X$ is regular at $p$, then WLOG, we can conclude that the map
\[
(u, v)\xmapsto{\ X\ }\big(x(u,v), y(u,v), z(u,v)\big)\xmapsto{\ \pi\ }\big(x(u,v), y(u,v)\big)
\]
is invertible on some neighborhood of $X^{-1}(p)$. Hence,
\[
\psi = X^{-1}\circ Y = \underbrace{(\pi\circ X)^{-1}}_{\text{differentiable}}\circ\underbrace{(\pi\circ Y)}_{\text{differentiable}}
\]
is differentiable. Similarly, $\psi^{-1}$ is differentiable. Therefore, by the bijectivity of $X, Y$, $\psi$ is a differentiable. 
\end{proof}

\definition{Smooth Function on Regular Surface}\\
Let $S$ be a regular surface and let $f: S\to\mathbb{R}$ be a function. We say $f$ is \textit{smooth} at $p\in S$ if $\exists$ coordinate patch $X: U\to V\cap S$ with $p\in X(U)$, s.t.: $f\circ X$ is smooth at $q = X^{-1}(p)\in U$.

\noindent\textbf{Remark}:
\begin{itemize}
\item In fact, for any parametrization $Y: U'\to V'\cap S$ with $p\in Y(U')$,
\[
f\circ Y = (f\circ X)\circ\big(X^{-1}\circ Y\big)
\]
is smooth on $Y^{-1}\big(X(U)\cap Y(U')\big)$.
\item One can extend this definition:
\begin{itemize}
\item to the function $f: S\to\mathbb{R}^n$, and
\item to $f: S_1\to S_2$ to be \textit{smooth} if $\exists$ $X_1, X_2$ coordinates patches of $S_1$ respectively $S_2$, such that

\[
\begin{tikzcd}
& X_1(U_1) \ar[rr, "f"] & & X_2(U_2) \\
U_1 \ar[ur, "X_1"] \ar[rrrr, "X_2^{-1}\circ f\circ X_1\ \text{is smooth}"] & & & & U_2 \ar[lu, "X_2"']
\end{tikzcd}
\]

\end{itemize}
\end{itemize}

\noindent\textbf{Example}: If $f: \mathbb{R}^3\to\mathbb{R}$ is smooth, then $f|_S: S\to\mathbb{R}$ is smooth, since $\forall X: U\to S\cap V$, $f\circ X: (u,v)\mapsto f\big(x(u,v), y(u,v), z(u,v)\big)$ is a composition of smooth functions.

\begin{proposition}{Double Cone is Not Regular}\\
Consider $F: \mathbb{R}^3\to\mathbb{R}$ given by $(x, y, z)\mapsto x^2 + y^2 - z^2$. Then $F^{-1}(0)$ is not a regular surface.
\end{proposition}

\begin{proof}
\underline{WARNING}: We can not just use $dF_p = (2x, 2y, 2z) = 0$ at $p = (0, 0, 0)$, since perhaps the surface $F^{-1}(0)$ can be described as a regular surface by other parametrizations.

\noindent Suppose on the contrary that $F^{-1}(0)$ is a regular surface. Then a neighborhood of $(0, 0, 0)$ can be described as one of the three functions:
\[
(x, y)\mapsto\big(x, y, f(x, y)\big), \quad (x, z)\mapsto\big(x, g(x, z), z\big), \quad\text{and}\quad (y, z)\mapsto\big(h(y, z), y, z\big)
\]
WLOG, let the first one describe locally $F^{-1}(0)$ at $(0, 0, 0)$. $\forall$ open ball $V = B(0, \varepsilon)$,
\[
(x_0, y_0, z_0)\in V\cap S \iff (x_0, y_0, -z_0)\in V\cap S.
\]
Hence, $f$ is not a function, contradiction. 
\end{proof}


\subsection{Tangent Plane}

\definition{Total Derivative}\\
Let $p\in S$ be in the image of two parametrizations $X_1: U\to V\cap S$, $X_2: \tilde{U}\to\tilde{V}\cap S$. Then the \textit{total derivative} at $p$ is defined as:
\[
d\psi := \begin{pmatrix}\dfrac{\partial\tilde{u}}{\partial u} & \dfrac{\partial\tilde{u}}{\partial v}\\[3mm] \dfrac{\partial\bar{v}}{\partial u} & \dfrac{\partial\bar{v}}{\partial v}\end{pmatrix} = \frac{\partial(\tilde{u}, \bar{v})}{\partial(u, v)}
\]
which is the Jacobian matrix of the transition map $\psi = X_2^{-1}\circ X_1: X_1^{-1}(W)\to X_2(W)$ where $W := X_1(U)\cap X_2(\tilde{U})\ni p$.

\noindent\textbf{Rmk}: By the change of coordinates Prop., $\psi$ is a diffeomorphism, so $\big(d\psi\big)_p$ is invertible.

\begin{proposition}
\[
\frac{\partial X_1}{\partial u} = \frac{\partial\tilde{u}}{\partial u}\frac{\partial X_2}{\partial\tilde{u}} + \frac{\partial\bar{v}}{\partial u}\frac{\partial X_2}{\partial\bar{v}} \quad\text{and}\quad \frac{\partial X_1}{\partial v} = \frac{\partial\tilde{u}}{\partial v}\frac{\partial X_2}{\partial\tilde{u}} + \frac{\partial\bar{v}}{\partial v}\frac{\partial X_2}{\partial\bar{v}}.
\]
\end{proposition}

\begin{proof}
It's immediately from the fact that $\forall(u, v)\in X_1^{-1}(W)$, $X_1(u, v) = X_2\big(\tilde{u}(u, v), \bar{v}(u, v)\big)$.
\end{proof}

\begin{corollary}
\[
\mathrm{Span}\left\{\frac{\partial X_1}{\partial u}, \frac{\partial X_1}{\partial v}\right\} = \mathrm{Span}\left\{\frac{\partial X_2}{\partial\tilde{u}}, \frac{\partial X_2}{\partial\bar{v}}\right\}.
\]
\end{corollary}

\noindent\textbf{Rmk}: This coro. gives the well-defined-ness of tangent plane (we'll introduce it later), i.e.: the tangent plane is not affected by using different coordinate patches.

\definition{Tangent Plane}\\
Let $X: U\to V\cap S$ be a coordinate patch of a regular surface at a point $X(q) = p\in S$. Then the \textit{tangent plane} at $p\in S$ is defined as:
\[
T_pS := \mathrm{Im}\big(dX_p\big) = \mathrm{Span}\left\{\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right\}.
\]

\noindent\textbf{Example}: $S^2$. A coordinate patch is given by $X(u, v) = \big(u, \sqrt{1 - u^2 - v^2}, v\big)$. Then for $p = \left(\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right) = X\left(\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right)$, one has
\[
dX_p = \begin{pmatrix}1 & 0\\ -\frac{u}{\sqrt{1-u^2-v^2}} & -\frac{v}{\sqrt{1-u^2-v^2}}\\ 0 & 1\end{pmatrix}
\Rightarrow T_p\big(S^2\big) = \mathrm{Span}\left\{\begin{pmatrix}1\\ -\frac{1}{\sqrt{3}}\\ 0\end{pmatrix}, \begin{pmatrix}0\\ -\frac{1}{\sqrt{3}}\\ 1\end{pmatrix}\right\} = \mathrm{Span}\left\{\begin{pmatrix}1\\ -1\\ 0\end{pmatrix}, \begin{pmatrix}0\\ -1\\ 1\end{pmatrix}\right\}.
\]

\noindent\begin{itemize}
\item For a (local) graph, $X(u, v) = (u, v, f(u, v))$, $p\in S$ with $p = X(u_0, v_0)$,
\[
T_pS = \mathrm{Span}\left\{\begin{pmatrix}1\\ 0\\ \frac{\partial f}{\partial u}(u_0, v_0)\end{pmatrix}, \begin{pmatrix}0\\ 1\\ \frac{\partial f}{\partial v}(u_0, v_0)\end{pmatrix}\right\}
\]
\end{itemize}
\noindent or $\langle x - p, n\rangle = 0$, where $n$ is normal to $T_pS$.

\definition{Normal Vector}\\
The \textit{normal vector} of a point $p$ on a coordinate patch $X: U\to V\cap S$ is defined as:
\[
N(p) := \frac{\dfrac{\partial X}{\partial u}\times\dfrac{\partial X}{\partial v}}{\big\|\dfrac{\partial X}{\partial u}\times\dfrac{\partial X}{\partial v}\big\|}\ (q)
\]
where $q := X^{-1}(p)$.

\noindent\textbf{Example}: $S^2$.
\[
N\left(\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right) = \frac{\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}\times\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}}{\big\|\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}\times\begin{pmatrix}\frac{1}{\sqrt{3}}\\ \frac{1}{\sqrt{3}}\end{pmatrix}\big\|} = \left(-\frac{1}{\sqrt{3}}, -\frac{1}{\sqrt{3}}, \frac{1}{\sqrt{3}}\right)
\]

\noindent\begin{itemize}
\item Local graph:
\[
N(p) = \frac{\left(-\frac{\partial f}{\partial u}, -\frac{\partial f}{\partial v}, 1\right)}{\big\|\left(-\frac{\partial f}{\partial u}, -\frac{\partial f}{\partial v}, 1\right)\big\|}
\]
\end{itemize}

\noindent\textbf{Remark}: Observe that $N(p)$ can be ``$+$'' or ``$-$'' along the same direction, so it may be affected by $X$ up to $\pm 1$. This problem will be fixed by the ``orientability''.

\definition{Tangent Vector}\\
A \textit{tangent vector} at $p\in S$ is the tangent vector $\alpha'(0)$ of a regular curve $\alpha: (-\varepsilon, \varepsilon)\to S\subseteq\mathbb{R}^3$ with $\alpha(0) = p$.

\begin{proposition}
The set of tangent vectors at $p\in S$ is precisely equal to $T_pS$.
\end{proposition}

\begin{proof}
($\subseteq$): Let $\alpha: (-\varepsilon, \varepsilon)\to S$ be a regular curve with $\alpha(0) = p$. Since regular surfaces are locally graphs, then $\exists\ Y: U\to V\cap S$, s.t.:
\[
Y(u, v) = (u, v, f(u, v)) \quad\text{or}\quad (u, g(u, v), v) \quad\text{or}\quad \big(h(u, v), u, v\big).
\]
WLOG, let $Y(u, v) = (u, v, f(u, v))$. By taking $\varepsilon$ to be small enough, we may assume $\alpha\big((-\varepsilon, \varepsilon)\big)\subseteq V\cap S$. Now define
\[
\beta = \left((- \varepsilon, \varepsilon)\xrightarrow{\ \alpha\ }V\cap S\xrightarrow{\ Y^{-1}\ }U\subseteq\mathbb{R}^2\right)
\]
where $Y^{-1}$ is differentiable. Hence, $\alpha = Y\circ\beta = X\circ\big(X^{-1}\circ Y\circ\beta\big) = X\circ(\psi\circ\beta)$ where $\alpha, X, \psi$, and $\beta$ are all differentiable. Therefore,
\[
\alpha'(0) = d\alpha_0 = dX_q\circ d(\psi\circ\beta)_0 = \left(\frac{\partial X}{\partial u}, \frac{\partial X}{\partial v}\right)(q)\cdot\begin{pmatrix}a\\ b\end{pmatrix}\in\mathrm{Span}\left\{\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right\} = T_pS.
\]

\noindent ($\supseteq$): $\forall w\in T_pS$, let $w = \left(\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right)\begin{pmatrix}a\\ b\end{pmatrix}$ by the def of tangent plane, for some $a, b\in\mathbb{R}$. Then define $\gamma: (-\varepsilon, \varepsilon)\to\mathbb{R}^2$ by $t\mapsto q + t\begin{pmatrix}a\\ b\end{pmatrix}$ and then define $\alpha := X\circ\gamma$. Then we have:
\[
\alpha(0) := X\big(\gamma(0)\big) = X(q) = p \quad\text{and}\quad \alpha'(0) = dX_q\circ d\gamma_0 := \left(\frac{\partial X}{\partial u}(q), \frac{\partial X}{\partial v}(q)\right)\cdot\begin{pmatrix}a\\ b\end{pmatrix}\Bigg|_{t=0} = w.
\]
\end{proof}

\noindent\textbf{Example}: Suppose $f: \mathbb{R}^3\to\mathbb{R}$ is a smooth function and $S = f^{-1}(a)$ where $a$ is a regular value of $f$. Then $S$ is a regular surface. Let $\alpha: (-\varepsilon, \varepsilon)\to S$ be any regular curve on $S$, with $\alpha(0) = p$. Then
\[
f\big(\alpha(t)\big)\in f\big(f^{-1}(a)\big) = \{a\} \Rightarrow f\big(\alpha(t)\big) = a, \quad \forall t\in(-\varepsilon, \varepsilon).
\]
Hence, $df_p\circ d\alpha_0 = 0$, i.e.: $df_p\cdot\alpha'(0) = 0$. Therefore, $df_p$ is normal to $T_pS$.

\subsection{First Fundamental Form (f.f.f.)}

\definition{First Fundamental Form (f.f.f.)}\\
The \textit{f.f.f.} of $S$ is given by the quadratic form $I_p(v) := \langle v, v\rangle$ for $v\in T_pS$.

\noindent\textbf{Remark}: Let $X: U\to V\cap S$ be a coordinate patch with $X(q) = p$. Then
\[
\mathcal{B} = \left\{X_u := \frac{\partial X}{\partial u}(q), X_v := \frac{\partial X}{\partial v}(q)\right\}
\]
is a basis of $T_pS$. If we write $E_p := \langle X_u, X_u\rangle_p$, $F_p := \langle X_u, X_v\rangle_p$ and $G_p := \langle X_v, X_v\rangle_p$ (for brevity, we often drop ``$p$''), then $\forall v, w\in T_pS$, $\exists\ a, b, c, d\in\mathbb{R}$, s.t.:
\[
\langle v, w\rangle_p = \langle aX_u + bX_v, cX_u + dX_v\rangle_p
\]
\[
= ac\,\langle X_u, X_u\rangle_p + (bc + ad)\,\langle X_u, X_v\rangle_p + bd\,\langle X_v, X_v\rangle_p
\]
\[
= ac\,E + (bc + ad)\,F + bd\,G
\]
\[
= \begin{pmatrix}a & b\end{pmatrix}\begin{pmatrix}E & F\\ F & G\end{pmatrix}\begin{pmatrix}c\\ d\end{pmatrix}.
\]
Hence, $\forall v\in T_pS$, $I_p(v) = \begin{pmatrix}a & b\end{pmatrix}\begin{pmatrix}E & F\\ F & G\end{pmatrix}\begin{pmatrix}a\\ b\end{pmatrix}$ where $v = aX_u + bX_v$. Therefore, we can retrieve $I_p$ from the matrix
\[
\begin{pmatrix}E & F\\ F & G\end{pmatrix} := \begin{pmatrix}\langle X_u, X_u\rangle_p & \langle X_u, X_v\rangle_p\\ \langle X_v, X_u\rangle_p & \langle X_v, X_v\rangle_p\end{pmatrix}.
\]
We then will often call this matrix the f.f.f. as well.

\noindent\textbf{Example}:
\begin{itemize}
\item \textit{Cylinder}. $X(u, v) = (\cos u, \sin u, v)$, $0 < u < 2\pi$, $v\in\mathbb{R}$.
\[
\Rightarrow X_u = (-\sin u, \cos u, 0) \quad\text{and}\quad X_v = (0, 0, 1).
\]
\[
\Rightarrow E = \langle X_u, X_u\rangle = \sin^2 u + \cos^2 u = 1, \quad F = \langle X_u, X_v\rangle = 0, \quad\text{and}\quad G = \langle X_v, X_v\rangle = 1.
\]
\[
\Rightarrow I_p = \begin{pmatrix}E & F\\ F & G\end{pmatrix} = \begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}.
\]
\item \textit{Plane}. $X(u, v) = x_0 + uW_1 + vW_2 \Rightarrow I_p = \begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix}$
\item \textit{Sphere}. $X(u, v) = \big(u, v, \sqrt{1 - u^2 - v^2}\big)$
\[
\Rightarrow X_u = \left(1, 0, \frac{-u}{\sqrt{1 - u^2 - v^2}}\right) \quad\text{and}\quad X_v = \left(0, 1, \frac{-v}{\sqrt{1 - u^2 - v^2}}\right)
\]
\[
\Rightarrow E = 1 + \frac{u^2}{1 - u^2 - v^2} = \frac{1 - v^2}{1 - u^2 - v^2}, \quad F = \frac{uv}{1 - u^2 - v^2}, \quad\text{and}\quad G = \frac{1 - u^2}{1 - u^2 - v^2}
\]
\[
\Rightarrow I_p = \begin{pmatrix}\dfrac{1 - v^2}{1 - u^2 - v^2} & \dfrac{uv}{1 - u^2 - v^2}\\[3mm] \dfrac{uv}{1 - u^2 - v^2} & \dfrac{1 - u^2}{1 - u^2 - v^2}\end{pmatrix}
\]
\end{itemize}

\noindent On the other hand, consider another coordinate patch $\tilde{X}(\theta, \phi) := (\sin\theta\cos\phi, \sin\theta\sin\phi, \cos\theta)$.
\[
\Rightarrow \tilde{X}_\theta = (\cos\theta\cos\phi, \cos\theta\sin\phi, -\sin\theta) \quad\text{and}\quad \tilde{X}_\phi = (-\sin\theta\sin\phi, \sin\theta\cos\phi, 0).
\]
\[
\Rightarrow \tilde{E} = 1, \quad \tilde{F} = 0, \quad\text{and}\quad \tilde{G} = \sin^2\theta \Rightarrow I_p = \begin{pmatrix}1 & 0\\ 0 & \sin^2\theta\end{pmatrix}
\]

\noindent Therefore, it's natural to ask: if we have two parametrizations $X: U\to V\cap S$ and $\tilde{X}: \tilde{U}\to\tilde{V}\cap S$, then how are their f.f.f.s related?

\begin{proposition}
Let $\psi = X^{-1}\circ\tilde{X}$ be the transition map. Then
\[
\begin{pmatrix}\tilde{E} & \tilde{F}\\ \tilde{F} & \tilde{G}\end{pmatrix} = \big(d\psi\big)^t\begin{pmatrix}E & F\\ F & G\end{pmatrix}\big(d\psi\big)
\]
i.e.: the related f.f.f.s are conjugated by the Jacobian matrix of $\psi$.
\end{proposition}

\begin{proof}
\begin{align*}
\begin{pmatrix}\tilde{E} & \tilde{F}\\ \tilde{F} & \tilde{G}\end{pmatrix} &= \left(\frac{\partial\tilde{X}}{\partial\tilde{u}}, \frac{\partial\tilde{X}}{\partial\bar{v}}\right)^{\!\!t}\left(\frac{\partial\tilde{X}}{\partial\tilde{u}}, \frac{\partial\tilde{X}}{\partial\bar{v}}\right) = \big(d\tilde{X}\big)^t\big(d\tilde{X}\big) = \big(dX\circ d\psi\big)^t\big(dX\circ d\psi\big)\\
&= \big(d\psi\big)^t\Big(\big(dX\big)^t\big(dX\big)\Big)\big(d\psi\big) = \big(d\psi\big)^t\begin{pmatrix}E & F\\ F & G\end{pmatrix}\big(d\psi\big).
\end{align*}
\end{proof}

\begin{corollary}
\[
\sqrt{\tilde{E}\tilde{G} - \tilde{F}^2} = \sqrt{EG - F^2}\cdot\left|\det\left(\frac{\partial(u, v)}{\partial(\tilde{u}, \bar{v})}\right)\right|
\]
\end{corollary}

\begin{proof}
It's immediately from the fact that $\psi: (\tilde{u}, \bar{v})\mapsto\big(u(\tilde{u}, \bar{v}), v(\tilde{u}, \bar{v})\big)$ gives the Jacobian matrix
\[
d\psi = \frac{\partial(u, v)}{\partial(\tilde{u}, \bar{v})} \Rightarrow \det\big(d\psi\big) = \det\big((d\psi)^t\big) = \det\left(\frac{\partial(u, v)}{\partial(\tilde{u}, \bar{v})}\right).
\]
\end{proof}

\subsection{Arc-length and Area}

\subsubsection{Arc-length of Regular Curve on Regular Surface}

\begin{proposition}
Let $\alpha: I\to U\subseteq\mathbb{R}^2$ be a regular curve on $U$, let $S\subseteq\mathbb{R}^3$ be a regular surface, and let $X: U\to V\cap S$ be a coordinate patch. Then the curve $\beta = X\circ\alpha: I\to S$ has length
\begin{align*}
    \int_I\big|\beta'(t)\big|\,dt &= \int_I\sqrt{\Big\langle\frac{d(X\circ\alpha)}{dt}, \frac{d(X\circ\alpha)}{dt}\Big\rangle}\ dt = \int_I\sqrt{\big(dX\cdot d\alpha\big)^t\big(dX\cdot d\alpha\big)}\ dt\\
    &= \int_I\sqrt{\big(x'(t)\ y'(t)\big)\big(dX\big)^t dX\begin{pmatrix}x'(t)\\ y'(t)\end{pmatrix}}\ dt = \int_I\sqrt{\big(x'(t)\ y'(t)\big)\,I_p\begin{pmatrix}x'(t)\\ y'(t)\end{pmatrix}}\ dt\\
    &= \int_I\sqrt{E\big(x'(t)\big)^2 + G\big(y'(t)\big)^2 + 2F\,x'(t)\,y'(t)}\ dt
\end{align*}
\end{proposition}

\subsubsection{Area of Regular Surface}

\definition{Area}\\
Let $X: U\to V\cap S$ be a coordinate patch of $S$ and let $R\subseteq S$ be such that $X(Q) = R$ for some $Q\subseteq U$. Then the \textit{area} of $R$ is given by:
\[
\int_Q d\sigma = \int_Q\big|X_u\times X_v\big|\,du\,dv
\]

\noindent\textbf{Remark}: Note that
\[
\big|X_u\times X_v\big|^2 = |X_u|^2|X_v|^2\sin^2\theta = |X_u|^2|X_v|^2\big(1 - \cos^2\theta\big) = |X_u|^2|X_v|^2 - \langle X_u, X_v\rangle^2
\]
Hence, $\big|X_u\times X_v\big| = \sqrt{EG - F^2}$. If $\tilde{X}: \tilde{U}\to\tilde{V}\cap S$ is another coordinate patch of $S$, and $\psi := X^{-1}\circ\tilde{X}: \tilde{Q}\to Q$ is a transition map, then
\[
\int_{\tilde{Q}}\sqrt{\tilde{E}\tilde{G} - \tilde{F}^2}\ d\tilde{u}\,d\tilde{v} = \int_{\tilde{Q}}\sqrt{EG - F^2}\cdot\left|\frac{\partial(u, v)}{\partial(\tilde{u}, \tilde{v})}\right|d\tilde{u}\,d\tilde{v} = \int_Q\sqrt{EG - F^2}\,du\,dv
\]
Therefore, the area of $R$ is independent of the choice of coordinate patches.

\noindent\textbf{Example}:
\begin{itemize}
\item \textit{Cylinder}. $X(u, v) = (\cos u, \sin u, v)$, $0 < u < 2\pi$, $v\in(a, b)$. Then $I_p = \begin{pmatrix}1 & 0\\ 0 & 1\end{pmatrix} \Rightarrow EG - F^2 = 1$. Hence,
\[
\text{the area of } X(u, v) = \int_a^b\int_0^{2\pi}1\,du\,dv = 2\pi(b - a).
\]
\item \textit{Sphere}. $X(\theta, \phi) = (\sin\theta\cos\phi, \sin\theta\sin\phi, \cos\theta)$ with $0 < \theta\le\pi$, $0 < \phi < 2\pi$ covers the sphere except for poles. Hence,
\[
\text{the area of } S^2 = \int_0^\pi\int_0^{2\pi}\sqrt{\sin^2\theta}\ d\phi\,d\theta = 4\pi.
\]
\item \textit{Torus}. $X(u, v) = \big((R + r\cos v)\cos u, (R + r\cos v)\sin u, r\sin v\big)$, with $0 < u < 2\pi$, $0 < v < 2\pi$. Then
\[
X_u = \big(-(R + r\cos v)\sin u, (R + r\cos v)\cos u, 0\big), \quad X_v = \big(-r\sin v\cos u, -r\sin v\sin u, r\cos v\big)
\]
Hence,
\[
\begin{cases}
E = \langle X_u, X_u\rangle = (R + r\cos v)^2\sin^2 u + (R + r\cos v)^2\cos^2 u = (R + r\cos v)^2\\[2mm]
F = \langle X_u, X_v\rangle = (R + r\cos v)\sin v\cos u\sin u - (R + r\cos v)\sin v\cos u\sin u + 0 = 0\\[2mm]
G = \langle X_v, X_v\rangle = r^2\sin^2 v\cos^2 u + r^2\sin^2 v\sin^2 u + r^2\cos^2 v = r^2
\end{cases}
\]
Therefore,
\[
I_p = \begin{pmatrix}(R + r\cos v)^2 & 0\\ 0 & r^2\end{pmatrix}
\]
so $\sqrt{EG - F^2} = \sqrt{(R + r\cos v)^2\,r^2} = r(R + r\cos v) \Rightarrow \text{Area of } T^2 = \int_0^{2\pi}\int_0^{2\pi}r(R + r\cos v)\,du\,dv = 4\pi^2 Rr.
$
\end{itemize}

\section{Gauss Map}

\subsection{Orientability}

\noindent\textbf{Goal}: We want to define the curvature of a regular surface. Recall that in the case of curve, in particular, of plane curve,
\[
T' = KN \quad\text{and}\quad N' = -KT
\]
Note that we need a \underline{global} concept, normal vector. We deal with this problem by definition:

\definition{Orientable Surface}\\
A surface $S$ is called \textit{orientable} if there is a smooth map $N: S\to S^2$, s.t.: $N(p)\perp T_pS$, $\forall p\in S$.

\noindent\textbf{Remark}: The codomain of the smooth map is $S^2$ instead of $\mathbb{R}^3$, which is because the normal vector is unit.

\noindent\textbf{Example}: 
\noindent\begin{itemize}
\item If $S$ is covered by exactly one coordinate patch $X: U\to S$, then
\[
N(p) = \frac{X_u\times X_v}{\|X_u\times X_v\|}\ (q) \quad\text{where } q = X^{-1}(p)
\]
e.g.: every graph is orientable.
\item Let $f: \mathbb{R}^3\to\mathbb{R}$ be a smooth function, and let $a\in\mathbb{R}$ be a regular value. Then consider $F := f - a$, so $S = F^{-1}(0)$ is a regular surface. Hence,
\[
N(p) := dF_p = \left(\frac{\partial F}{\partial x}, \frac{\partial F}{\partial y}, \frac{\partial F}{\partial z}\right)(p)
\]
\end{itemize}
